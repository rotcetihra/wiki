Транзакции позволяют выполнять набор операций в виде одного целостного пакета. Если хотя бы одна из операций завершится неудачно, произойдет откат выполнения остальных операций.

```csharp
using (SqlConnection connection = new SqlConnection(connectionString))
{
    await connection.OpenAsync();
    SqlTransaction transaction = connection.BeginTransaction();

    SqlCommand command = connection.CreateCommand();
    command.Transaction = transaction;

    try
    {
        command.CommandText = "INSERT INTO Users (Name, Age) VALUES('Tim', 34)";
        await command.ExecuteNonQueryAsync();
        command.CommandText = "INSERT INTO Users (Name, Age) VALUES('Kat', 31)";
        await command.ExecuteNonQueryAsync();

        await transaction.CommitAsync();
        Console.WriteLine("Данные добавлены в базу данных");
    }
    catch (Exception ex)
    {
        Console.WriteLine(ex.Message);
        await transaction.RollbackAsync();
    }
}
```

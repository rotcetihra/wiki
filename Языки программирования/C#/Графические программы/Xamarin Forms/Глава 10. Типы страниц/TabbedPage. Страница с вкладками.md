Также изменим в коде страницы ее тип с ContentPage на TabbedPage:

```

using Xamarin.Forms;

namespace HelloApp
{
    public partial class MainPage : TabbedPage
    {
        public MainPage()
        {
            InitializeComponent();
        }
    }
}

```

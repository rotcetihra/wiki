## Операторы преобразования типов

C++ позволяет определить классы, которые функционируют как другие типы. Такой подход будет полезен, например, если у нас определены классы для различных типов фигур и tenemos функцию, которая принимает в качестве аргумента объект класса Point, который определяет координаты на плоскости. В эту функцию мы также можем передать объект круга или прямоугольника, если определим операторы преобразования типов.

```cpp
#include <iostream>

class Point
{
public:
    int x;
    int y;
    Point(int x, int y): x{x}, y{y}{}
};

class Circle
{
public:
    Circle(int r): radius{r}{}
    operator Point() const
    {
        std::cout << "Circle is converted to point" << std::endl;
        return Point{radius, radius};
    }
private:
    int radius;
};

void printPoint(const Point& p)
{
    std::cout << "x: " << p.x << "\t y: " << p.y << std::endl;
}

int main()
{
    Circle circle{10};
    Point p = circle;
    printPoint(p);  // Circle is converted to point
                    // x: 10      y: 10
}
```

### Множественные преобразования

```cpp
#include <iostream>
#include <cmath>

class Point
{
public:
    int x;
    int y;
    Point(int x, int y): x{x}, y{y}{}
    double length() const
    {
        return std::sqrt(x*x + y*y);
    }
};

class Circle
{
public:
    Circle(double r): radius{r}{}
    operator Point() const
    {
        std::cout << "Circle is converted to point" << std::endl;
        return Point{(int)radius, 0};
    }
    bool operator<(const Point& p) const
    {
        return radius < p.length();
    }
private:
    double radius;
};

void printPoint(const Point& p)
{
    std::cout << "x: " << p.x << "\t y: " << p.y << std::endl;
}

int main()
{
    Circle circle{10};
    Point p{1, 2};
    printPoint(circle);     // Circle is converted to point
                            // x: 10      y: 0
    if(circle < p)
        std::cout << "circle is less than point" << std::endl;
}
```

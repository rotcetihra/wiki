## Вложенные классы

C++ позволяет создавать классы в других классах. Подобные классы называются вложенными классами:

```cpp
#include <iostream>

class Array
{
public:
    class Iterator
    {
    public:
        Iterator(int* p) : p_ptr{p}{ }
        int& operator*() {return *p_ptr;}
        void operator++() {p_ptr++;}
        bool operator!=(const Iterator& other) {return p_ptr != other.p_ptr;}
    private:
        int* p_ptr;
    };

    Array(unsigned size): size{size}
    {
        data = new int[size]{};
    }
    Iterator begin() {return Iterator{data};}
    Iterator end() {return Iterator{data + size};}
    int& operator[](unsigned index) {return data[index];}
    void print() const
    {
        for(unsigned i = 0; i < size; i++)
            std::cout << data[i] << " ";
        std::cout << std::endl;
    }
    ~Array() { delete[] data; }
private:
    unsigned size;
    int* data;
};

int main()
{
    Array numbers{5};
    numbers[0] = 10;
    numbers[1] = 20;
    numbers[2] = 30;
    numbers[3] = 40;
    numbers[4] = 50;

    Array::Iterator it{numbers.begin()};
    for(Array::Iterator it = numbers.begin(); it != numbers.end(); ++it)
        std::cout << *it << " ";
    std::cout << std::endl;

    for(auto n : numbers)
        std::cout << n << " ";
}
```

Класс Iterator определен внутри класса Array и используется для перебора элементов контейнера. При его определении указывается Array::Iterator.

### Преимущество вложенных классов

Вложенный класс получает доступ к приватным членам внешнего класса:

```cpp
class Array
{
public:
    class Iterator
    {
    public:
        Iterator(int* p) : p_ptr{p}{ }
        int& operator*() {return *p_ptr;}
        void operator++() {p_ptr++;}
        bool operator!=(const Iterator& other) {return p_ptr != other.p_ptr;}
    private:
        int* p_ptr;
    };
    // ...
private:
    unsigned size;
    int* data;
};
```

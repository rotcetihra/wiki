## Переопределение оператора присваивания

Оператор присваивания = позволяет копировать содержимое одного объекта в другой объект того же типа. По умолчанию компилятор создает собственную версию оператора присваивания, которая копирует каждую переменную. В большинстве случаев этого достаточно. Однако возможны ситуации, когда этого недостаточно.

```cpp
#include <iostream>

class Array
{
public:
    Array(unsigned size): size{size}
    {
        data = new int[size]{};
    }
    Array(const Array& other): size{other.size}
    {
        data = new int[size]{};
        for(unsigned i = 0; i < size; i++)
            data[i] = other.data[i];
    }
    Array& operator=(const Array& other)
    {
        if(this != &other)
        {
            delete[] data;
            size = other.size;
            data = new int[size]{};
            for(unsigned i = 0; i < size; i++)
                data[i] = other.data[i];
        }
        return *this;
    }
    int& operator[](unsigned index) {return data[index];}
    void print() const
    {
        for(unsigned i = 0; i < size; i++)
            std::cout << data[i] << " ";
        std::cout << std::endl;
    }
    ~Array() { delete[] data; }
private:
    unsigned size;
    int* data;
};

int main()
{
    Array numbers1{3};
    numbers1[0] = 1;
    numbers1[1] = 2;
    numbers1[2] = 3;
    Array numbers2{2};
    numbers2[0] = 5;
    numbers2[1] = 4;

    numbers1 = numbers2;
    numbers1.print();   // 5 4 0
}
```

В Vulkan шейдеры инкапсулированы в шейдерные модули, которые загружаются в GPU и могут использоваться для создания конвейеров. Прежде чем передать код шейдера в конвейер, его нужно обернуть в шейдерный модуль - объект `VkShaderModule`.

Для создания модуля шейдера применяется функция `vkCreateShaderModule()`:

```cpp
VkResult vkCreateShaderModule(
    VkDevice                                    device,
    const VkShaderModuleCreateInfo*             pCreateInfo,
    const VkAllocationCallbacks*                pAllocator,
    VkShaderModule*                             pShaderModule);
```

Для создания модуля шейдера определяем структуру `VkShaderModuleCreateInfo`:

```cpp
typedef struct VkShaderModuleCreateInfo {
    VkStructureType              sType;
    const void*                  pNext;
    VkShaderModuleCreateFlags    flags;
    size_t                       codeSize;
    const uint32_t*              pCode;
} VkShaderModuleCreateInfo;
```

Поля структуры:

- `sType`: тип структуры - `VK_STRUCTURE_TYPE_SHADER_MODULE_CREATE_INFO`.
- `flags`: не используется и зарезервирован для будущего использования.
- `codeSize`: размер в байтах кода, на который указывает `pCode`.
- `pCode`: указатель на код, который используется для создания модуля шейдера.

Пример загрузки модуля:

```cpp
VkShaderModule createShaderModule(const std::vector<char>& code) {
    VkShaderModuleCreateInfo createInfo{};
    createInfo.sType = VK_STRUCTURE_TYPE_SHADER_MODULE_CREATE_INFO;
    createInfo.codeSize = code.size();
    createInfo.pCode = reinterpret_cast<const uint32_t*>(code.data());

    VkShaderModule shaderModule;
    if (vkCreateShaderModule(device, &createInfo, nullptr, &shaderModule) != VK_SUCCESS) {
        throw std::runtime_error("Не удалось создать модуль шейдера!");
    }
    return shaderModule;
}
```

Удаление модуля шейдера:

```cpp
vkDestroyShaderModule(device, shaderModule, nullptr);
```

## Применение модулей шейдеров

Чтобы использовать шейдеры, нужно передать их на определенный этап конвейера через структуру `VkPipelineShaderStageCreateInfo`:

```cpp
VkPipelineShaderStageCreateInfo vertShaderStageInfo{};
vertShaderStageInfo.sType = VK_STRUCTURE_TYPE_PIPELINE_SHADER_STAGE_CREATE_INFO;
vertShaderStageInfo.stage = VK_SHADER_STAGE_VERTEX_BIT;
vertShaderStageInfo.module = vertShaderModule;
vertShaderStageInfo.pName = "main";

VkPipelineShaderStageCreateInfo fragShaderStageInfo{};
fragShaderStageInfo.sType = VK_STRUCTURE_TYPE_PIPELINE_SHADER_STAGE_CREATE_INFO;
fragShaderStageInfo.stage = VK_SHADER_STAGE_FRAGMENT_BIT;
fragShaderStageInfo.module = fragShaderModule;
fragShaderStageInfo.pName = "main";

VkPipelineShaderStageCreateInfo shaderStages[] = {vertShaderStageInfo, fragShaderStageInfo};
```

Очереди отвечают за выполнение буферов команд, которые содержат инструкции для GPU. Очереди автоматически создаются вместе с логическим устройством. На уровне кода очереди представляют непрозрачный указатель (opaque handle) `VkQueue`. Соответственно для работы с очередями нам сначала надо определить переменную этого типа:

```cpp
VkQueue graphicsQueue;
```

Очереди устройств неявно очищаются при уничтожении устройства, поэтому нам не нужно ничего дополнительно делать.

Для получения очереди можно использовать функцию `vkGetDeviceQueue`:

```cpp
void vkGetDeviceQueue(
    VkDevice            device,
    uint32_t            queueFamilyIndex,
    uint32_t            queueIndex,
    VkQueue*            pQueue);
```

Функция принимает следующие параметры:

- `device`: логическое устройство, которому принадлежит очередь.
- `queueFamilyIndex`: индекс семейства очередей, к которому принадлежит очередь.
- `queueIndex`: индекс в этом семействе очередей очереди, которую нужно извлечь.
- `pQueue`: указатель на объект `VkQueue`, который будет заполнен дескриптором запрошенной очереди.

Если мы создаем только одну очередь для определенного семейства (например, нам нужна только графическая очередь для работы с графикой), то в качестве индекса очереди в этом семействе просто используем индекс 0:

```cpp
VkQueue graphicsQueue;
vkGetDeviceQueue(device, queueFamilyIndex, 0, &graphicsQueue);
```

Пример создания очереди:

```cpp
#include <vulkan/vulkan.h>
#include <iostream>
#include <vector>

// функция для получения первой видеокарты в списке устройств
VkPhysicalDevice selectFirstDevice(VkInstance* instance){

    uint32_t deviceCount = 0;
    vkEnumeratePhysicalDevices(*instance, &deviceCount, nullptr);
    if (!deviceCount)  return VK_NULL_HANDLE; 

    std::vector<VkPhysicalDevice> devices(deviceCount);
    vkEnumeratePhysicalDevices(*instance, &deviceCount, devices.data());

    return devices[0];
}

// выбираем семейство очередей с графической очередью
uint32_t selectQueueFamilyIndex(VkPhysicalDevice device){

    uint32_t familyCount = 0;
    vkGetPhysicalDeviceQueueFamilyProperties(device, &familyCount, nullptr);
    if (!familyCount)  return UINT32_MAX; 

    VkQueueFamilyProperties families[familyCount];
    vkGetPhysicalDeviceQueueFamilyProperties(device, &familyCount, families);

    for (int32_t i{}; i < familyCount; ++i) {
        if (families[i].queueFlags & VK_QUEUE_GRAPHICS_BIT) {
            return i;
        }
    }
    return UINT32_MAX;
}

// функция создания логического устройства
VkDevice createLogicalDevice(VkPhysicalDevice physicalDevice, uint32_t queueFamilyIndex){

    if(queueFamilyIndex == UINT32_MAX) return VK_NULL_HANDLE;

    VkDevice device = VK_NULL_HANDLE;

    VkDeviceQueueCreateInfo queueCreateInfo = {};
    queueCreateInfo.sType = VK_STRUCTURE_TYPE_DEVICE_QUEUE_CREATE_INFO;
    queueCreateInfo.queueFamilyIndex = queueFamilyIndex;
    queueCreateInfo.queueCount = 1;
    float queuePriority = 1.0f;
    queueCreateInfo.pQueuePriorities = &queuePriority;

    VkDeviceCreateInfo createInfo = {};
    createInfo.sType = VK_STRUCTURE_TYPE_DEVICE_CREATE_INFO;
    createInfo.queueCreateInfoCount = 1;
    createInfo.pQueueCreateInfos = &queueCreateInfo;

    vkCreateDevice(physicalDevice, &createInfo, nullptr, &device);
    return device;
}

// функция создания очереди для логического устройства
VkQueue createQueue(VkDevice device, uint32_t queueFamilyIndex){

    if(device== VK_NULL_HANDLE) return VK_NULL_HANDLE; 

    VkQueue graphicsQueue;
    vkGetDeviceQueue(device, queueFamilyIndex, 0, &graphicsQueue);
    return graphicsQueue;
}

int main() {
    
    VkInstance instance;
    VkInstanceCreateInfo createInfo{};
    createInfo.sType = VK_STRUCTURE_TYPE_INSTANCE_CREATE_INFO;

    if(vkCreateInstance(&createInfo, nullptr, &instance)!= VK_SUCCESS){
        std::cout << "Unable to create VKInstance" << std::endl;
        return 1;
    }
    VkPhysicalDevice physicalDevice = selectFirstDevice(&instance);
    uint32_t queueFamilyIndex = selectQueueFamilyIndex(physicalDevice);
    VkDevice device = createLogicalDevice(physicalDevice, queueFamilyIndex);

    VkQueue graphicsQueue = createQueue(device, queueFamilyIndex);
    std::cout << "Graphic Queue created" << std::endl;

    if(device!= VK_NULL_HANDLE) vkDestroyDevice(device, nullptr);

    vkDestroyInstance(instance, nullptr);
    return 0;
}
```

Здесь получение графической очереди вынесено в отдельную функцию `createQueue`, которая возвращает значение `VkQueue`:

```cpp
VkQueue createQueue(VkDevice device, uint32_t queueFamilyIndex){

    if(device== VK_NULL_HANDLE) return VK_NULL_HANDLE; 

    VkQueue graphicsQueue;
    vkGetDeviceQueue(device, queueFamilyIndex, 0, &graphicsQueue);
    return graphicsQueue;
}
```

В функции main вызываем эту функцию, проверяем создание очереди и выводим на консоль диагностическое сообщение:

```cpp
VkQueue graphicsQueue = createQueue(device, queueFamilyIndex);
std::cout << "Graphic Queue created" << std::endl;
```

Определив логическое устройство и очередь, мы теперь можем начать использовать графическую карту для выполнения различных задач.

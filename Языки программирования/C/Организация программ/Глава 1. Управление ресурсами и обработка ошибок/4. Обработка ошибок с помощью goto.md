## Описание

Функция, которая использует множество ресурсов и вынуждена выделять и освобождать для них память, может для управления ресурсами и обработки ошибок применять вложенные конструкции `if`. В подобных случаях выделение памяти для одного ресурса зависит от того, была ли ранее выделена память для другого ресурса. Однако если получение второго ресурса прошло неудачно, память первого ресурса всё равно надо освободить.

Использование вложенных `if` является естественным ответом на эту проблему, но затрудняет чтение и поддержку функции. Кроме того, образуется несколько мест в коде, где происходит освобождение ресурсов и обработка ошибок.

## Паттерн с goto

Одним из решений может быть вынесение кода освобождения ресурсов и обработки ошибок в конец функции в отдельную секцию, а в основном коде выполнять переход к этой секции с помощью оператора `goto` при возникновении ошибки:

```c
void some_function(void)
{
    if (!allocate_resource1()) {
        goto cleanup1;
    }
    if (!allocate_resource2()) {
        goto cleanup2;
    }

    // основной код

// секция освобождения ресурсов и обработки ошибок
cleanup2:
    cleanup_resource2();
cleanup1:
    cleanup_resource1();
}
```

Стоит отметить, что оператор `goto` часто не рекомендуется к применению в программировании, так как его употребление может вести к усложнению кода и снижению читабельности. В связи с этим различные стандарты предписывают не использовать данный оператор.

Тем не менее, конкретно в данном случае при должной организации мы можем уйти от чрезмерной вложенности `if` и логически отделить код обработки ошибок и освобождения ресурсов от основной логики программы. Функция будет иметь **одну точку выхода**, что упрощает управление её результатом. Данный паттерн довольно широко применяется в **ядре Linux** для обработки ошибок.

## Пример: поиск слова в файле

Рассмотрим программу, которая ищет определённое слово в файле. Изначально функция `find()` использует вложенные `if`:

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <assert.h>

typedef enum { ERROR, FOUND, NOTFOUND } status;

status find(char*, char*, size_t);
status find_word(FILE*, char*, size_t, char*);

status find(char *file_name, char *word, size_t word_size)
{
    FILE *file_pointer = 0;
    char *buffer = 0;
    status result = ERROR;

    assert(file_name != NULL && "File name is invalid!");
    assert(word != NULL && "Word is invalid!");

    if ((file_pointer = fopen(file_name, "r")))
    {
        if ((buffer = malloc(word_size)))
        {
            result = find_word(file_pointer, word, word_size, buffer);
            free(buffer);
        }
        fclose(file_pointer);
    }
    return result;
}

status find_word(FILE *file_pointer, char *word, size_t word_size, char *buffer)
{
    while (fgets(buffer, word_size, file_pointer))
    {
        if (strcmp(word, buffer) == 0)
        {
            return FOUND;
        }
    }
    return NOTFOUND;
}

int main(void)
{
    char *filename = "test.txt";
    char word[] = "hello";
    size_t size = sizeof(word);

    status result = find(filename, word, size);

    switch (result)
    {
        case ERROR:
            printf("Error!\n");
            break;
        case NOTFOUND:
            printf("Text not found...\n");
            break;
        case FOUND:
            printf("Success! Text found!\n");
            break;
    }
}
```

Функция `find()` проверяет предусловия с помощью `assert`, тем самым устраняя пару конструкций `if`. Управление ресурсами и основные действия отделены вынесением поиска в отдельную функцию `find_word`. Тем не менее, применение вложенных `if` снижает читабельность:

```c
if ((file_pointer = fopen(file_name, "r")))
{
    if ((buffer = malloc(word_size)))
    {
        // ...
    }
    // ...
}
```

## Применение goto

Используем `goto` для устранения вложенности:

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <assert.h>

typedef enum { ERROR, FOUND, NOTFOUND } status;

status find(char*, char*, size_t);
status find_word(FILE*, char*, size_t, char*);

status find(char *file_name, char *word, size_t word_size)
{
    FILE *file_pointer = 0;
    char *buffer = 0;
    status result = ERROR;

    assert(file_name != NULL && "File name is invalid!");
    assert(word != NULL && "Word is invalid!");

    // если НЕ удалось открыть файл
    if (!(file_pointer = fopen(file_name, "r"))) {
        goto error_fileopen;
    }

    // если память НЕ выделена
    if (!(buffer = malloc(word_size))) {
        goto error_malloc;
    }

    result = find_word(file_pointer, word, word_size, buffer);

    // секция освобождения ресурсов
    free(buffer);
error_malloc:
    fclose(file_pointer);
error_fileopen:
    return result;
}

status find_word(FILE *file_pointer, char *word, size_t word_size, char *buffer)
{
    while (fgets(buffer, word_size, file_pointer))
    {
        if (strcmp(word, buffer) == 0)
        {
            return FOUND;
        }
    }
    return NOTFOUND;
}

int main(void)
{
    char *filename = "test.txt";
    char word[] = "hello";
    size_t size = sizeof(word);

    status result = find(filename, word, size);

    switch (result)
    {
        case ERROR:
            printf("Error!\n");
            break;
        case NOTFOUND:
            printf("Text not found...\n");
            break;
        case FOUND:
            printf("Success! Text found!\n");
            break;
    }
}
```

## Порядок выполнения

Теперь функция `find` проверяет выделение ресурсов и в случае ошибки переходит к определённой метке в секции в конце функции:

```c
if (!(file_pointer = fopen(file_name, "r"))) {
    goto error_fileopen;
}
if (!(buffer = malloc(word_size))) {
    goto error_malloc;
}
```

В конце функции в отдельной секции происходит освобождение ресурсов и возвращение результата:

```c
    // секция освобождения ресурсов
    free(buffer);
error_malloc:
    fclose(file_pointer);
error_fileopen:
    return result;
```

| Сценарий | Переход | Действие |
|---|---|---|
| Файл открыт, буфер выделен | Нет | `free` → `fclose` → `return` |
| Буфер не выделен | `error_malloc` | `fclose` → `return` |
| Файл не открыт | `error_fileopen` | `return` |

> [!NOTE]
> Метки располагаются в **обратном порядке** относительно выделения ресурсов: последний выделенный ресурс освобождается первым. Это гарантирует, что каждый ресурс будет освобождён ровно один раз.

> [!WARNING]
> Оператор `goto` допустим только для выхода из блока, а не для перехода внутрь других блоков. Использование `goto` для перехода вперёд (в обход инициализации переменных) приводит к ошибкам компиляции.

## Описание

Иногда необходимо, чтобы функция уведомляла внешний контекст о результатах своей работы на различных её этапах. В этом случае распространённым решением является возврат из функции **статусных кодов**. Получив из функции статусный код, вызывающий код может его обработать по своему усмотрению.

В качестве статусного кода результата часто используются числовые значения. Например, в Windows Microsoft использует тип `HRESULT`, который фактически представляет 64-разрядное число и применяется для возвращения статусного кода результата.

## Пример: поиск слова в файле

Рассмотрим программу, которая ищет в файле определённое слово. Функция `find()` обрабатывает ошибки прямо на месте:

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void find(char*, char*, size_t);
void find_word(char*, char*, size_t, FILE*);

int main(void)
{
    char *filename = "test.txt";
    char word[] = "hello";
    size_t size = sizeof(word);
    find(filename, word, size);
}

void find_word(char *buffer, char *word, size_t word_size, FILE *file_pointer)
{
    int found = 0;
    while (fgets(buffer, word_size, file_pointer))
    {
        if (strcmp(word, buffer) == 0)
        {
            found = 1;
            break;
        }
    }
    if (found) {
        printf("Success! Word found!\n");
    } else {
        printf("Word not found...\n");
    }
}

void find(char *file_name, char *word, size_t word_size)
{
    FILE *file_pointer = 0;
    char *buffer = 0;

    if (!file_name) {
        printf("Invalid filename\n");
        return;
    }
    if (!word) {
        printf("Invalid word for search\n");
        return;
    }
    if (!(file_pointer = fopen(file_name, "r"))) {
        printf("Unable to open the file\n");
        return;
    }
    if ((buffer = malloc(word_size))) {
        find_word(buffer, word, word_size, file_pointer);
        free(buffer);
    } else {
        printf("Error! Unable to locate memory!\n");
    }
    fclose(file_pointer);
}
```

Здесь есть несколько узловых точек, каждая из которых обрабатывает определённое состояние: не указано имя файла, не указано слово, не удалось открыть файл, не удалось выделить память. В функции `find_word` для хранения результата пришлось создавать отдельную переменную `found`.

## Возврат статусных кодов через перечисление

Определим статусные коды для каждого состояния и вместо обработки на месте возвратим статусный код:

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef enum {
    FILENAME_INVALID,
    WORD_INVALID,
    FILEOPEN_ERROR,
    ALLOCATION_ERROR,
    WORD_FOUND,
    WORD_NOTFOUND
} status;

status find(char*, char*, size_t);
status find_word(char*, char*, size_t, FILE*);

int main(void)
{
    char *filename = "test.txt";
    char word[] = "hello";
    size_t size = sizeof(word);

    status result = find(filename, word, size);
    switch (result)
    {
        case FILENAME_INVALID:
            printf("Invalid filename\n");
            break;
        case WORD_INVALID:
            printf("Invalid word\n");
            break;
        case FILEOPEN_ERROR:
            printf("Unable to open the file\n");
            break;
        case ALLOCATION_ERROR:
            printf("Unable to allocate memory\n");
            break;
        case WORD_FOUND:
            printf("Success! Word found!\n");
            break;
        case WORD_NOTFOUND:
            printf("Word not found...\n");
            break;
    }
}

status find_word(char *buffer, char *word, size_t word_size, FILE *file_pointer)
{
    while (fgets(buffer, word_size, file_pointer))
    {
        if (strcmp(word, buffer) == 0)
        {
            return WORD_FOUND;
        }
    }
    return WORD_NOTFOUND;
}

status find(char *file_name, char *word, size_t word_size)
{
    FILE *file_pointer = 0;
    char *buffer = 0;
    status result;

    if (!file_name) {
        return FILENAME_INVALID;
    }
    if (!word) {
        return WORD_INVALID;
    }
    if (!(file_pointer = fopen(file_name, "r"))) {
        return FILEOPEN_ERROR;
    }
    if ((buffer = malloc(word_size))) {
        result = find_word(buffer, word, word_size, file_pointer);
        free(buffer);
    } else {
        result = ALLOCATION_ERROR;
    }
    fclose(file_pointer);
    return result;
}
```

Мы можем отделить логику обработки результата от основного действия функций `find`/`find_word`. И в то же время вызывающий код с помощью результата получит статус и должным образом обработает его.

## Обработка через таблицу

Вместо конструкции `switch` для обработки статуса можно использовать таблицы:

```c
int main(void)
{
    char *filename = "test.txt";
    char word[] = "hello";
    size_t size = sizeof(word);

    status result = find(filename, word, size);
    char *status_messages[6] =
    {
        "Invalid filename\n",
        "Invalid word\n",
        "Unable to open the file\n",
        "Unable to allocate memory\n",
        "Success! Word found!\n",
        "Word not found...\n"
    };
    printf(status_messages[result]);
}
```

Поскольку значение перечисления по сути представляет число (первая константа равна `0`, остальные увеличиваются на единицу), мы можем использовать статусный код в качестве индекса в массиве для получения нужного сообщения.

## Релевантность статусных кодов

Одна из проблем — насколько релевантны статусные коды для вызывающего кода. Возможно, для потребителя все ошибки аналогичны:

```c
int main(void)
{
    char *filename = "test21.txt";
    char word[] = "hello";
    size_t size = sizeof(word);

    status result = find(filename, word, size);

    switch (result)
    {
        case WORD_FOUND:
            printf("Success! Word found!\n");
            break;
        case WORD_NOTFOUND:
            printf("Word not found...\n");
            break;
        default:
            printf("Error occured...\n");
            break;
    }
}
```

Если статусные коды в принципе не будут использоваться, нет смысла перегружать функцию возвращением всех этих кодов. Достаточно вернуть только ту информацию, которая релевантна для потребителя:

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef enum {
    ERROR,
    WORD_FOUND,
    WORD_NOTFOUND
} status;

status find(char*, char*, size_t);
status find_word(FILE*, char*, size_t, char*);

int main(void)
{
    char *filename = "test.txt";
    char word[] = "hello";
    size_t size = sizeof(word);

    status result = find(filename, word, size);

    switch (result)
    {
        case WORD_FOUND:
            printf("Success! Word found!\n");
            break;
        case WORD_NOTFOUND:
            printf("Word not found...\n");
            break;
        default:
            printf("Error occured...\n");
            break;
    }
}

status find_word(FILE *file_pointer, char *word, size_t word_size, char *buffer)
{
    while (fgets(buffer, word_size, file_pointer))
    {
        if (strcmp(word, buffer) == 0)
        {
            return WORD_FOUND;
        }
    }
    return WORD_NOTFOUND;
}

status find(char *file_name, char *word, size_t word_size)
{
    FILE *file_pointer = 0;
    char *buffer = 0;
    status result = ERROR;

    if (file_name && word && (file_pointer = fopen(file_name, "r")) && (buffer = malloc(word_size)))
    {
        result = find_word(file_pointer, word, word_size, buffer);
    }

    if (buffer) free(buffer);
    if (file_pointer) fclose(file_pointer);

    return result;
}
```

Уменьшение диапазона возвращаемых статусных кодов кроме большей релевантности имеет побочный эффект — уменьшение и упрощение кода.

## Два состояния: успех или ошибка

Иногда не требуется сложная система статусных кодов — достаточно двух состояний: успех или ошибка. Часто можно встретить функции, которые возвращают `true`/`false`, `1`/`0`/`(-1)`:

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <assert.h>

int find(char*, char*, size_t);
int find_word(FILE*, char*, size_t, char*);

int main(void)
{
    char *filename = "test32.txt";
    char word[] = "hello";
    size_t size = sizeof(word);

    if (find(filename, word, size))
    {
        printf("Success! Word found!\n");
    } else {
        printf("Word not found...\n");
    }
}

int find_word(FILE *file_pointer, char *word, size_t word_size, char *buffer)
{
    while (fgets(buffer, word_size, file_pointer))
    {
        if (strcmp(word, buffer) == 0)
        {
            return 1;
        }
    }
    return 0;
}

int find(char *file_name, char *word, size_t word_size)
{
    FILE *file_pointer = 0;
    char *buffer = 0;
    int result = 0;

    assert(file_name != NULL && "File name is invalid!");
    assert(word != NULL && "Word is invalid!");

    if ((file_pointer = fopen(file_name, "r")) && (buffer = malloc(word_size)))
    {
        result = find_word(file_pointer, word, word_size, buffer);
    }

    if (buffer) free(buffer);
    if (file_pointer) fclose(file_pointer);

    return result;
}
```

Если слово найдено — возвращаем `1`, если нет (включая ошибки) — `0`. Это ещё больше сокращает и упрощает код.

## Сводка подходов

| Подход | Количество кодов | Когда использовать |
|---|---|---|
| Широкий набор кодов | 5–10+ | Вызывающий код обрабатывает каждую ошибку отдельно |
| Суженный набор | 2–3 | Ошибки группируются (`ERROR` / `FOUND` / `NOTFOUND`) |
| Два состояния | 2 | Достаточно знать «успех/неуспех» |

> [!NOTE]
> Между проектировщиком функции и её потребителем должно быть согласие насчёт того, какой именно результат и в каком виде должна возвращать функция. Чрезмерная детализация статусных кодов усложняет API без реальной выгоды.

В прошлых темах мы рассмотрели различные аспекты определения хеш-таблиц. Теперь посмотрим, как мы можем реализовать базовые операции над хеш-таблицей. Обычно к ним относят вставку и удаление ключей, а также запросы для проверки наличия заданного ключа в таблице. Для этих операций можно использовать следующий интерфейс:

```c
void insert_key(struct hash_table *table, unsigned int key);
bool contains_key(struct hash_table *table, unsigned int key);
void delete_key(struct hash_table *table, unsigned int key);
```

Для всех трех операций потребуется сама хеш-таблица и хеш-ключ данных, которые будут помещаться в хеш-таблицу.

Но сразу стоит отметить некоторые проблемы при проектировании, с которыми мы можем столкнуться. Прежде всего это отслеживание наличия данных в хеш-таблице. Например, данные ранее были добавлены, но потом были удалены, а потом они могут быть опять добавлены и т.д. В зависимости от конкретных условий и задачи реализации могут быть различными. Но обычно они представляют применение некоторого маркера, который указывает на наличие или отсутствие значения. Реализация подобного маркера также может быть различной.

Например, если корзина хеш-таблицы хранит только хеш-ключи, то в качестве маркера может использоваться некоторое недопустимое или невозможное значение, которое не может иметь хеш-ключ, например, 0 (если применяемая хеш-функция не генерирует подобного значения). Либо это может быть какое-то другое значение.

Другой, более распространенный вариант — хранить в структуре корзины специальное поле, которое будет указывать на наличие или отсутствие значения. Собственно далее я буду ориентироваться на этот вариант. И для этого в качестве примера возьмем следующие структуры:

```c
// структура корзины в хеш-таблице
struct bin
{
    int is_free;        // свободная ли корзина
    unsigned int key;   // хеш-ключ
};

// структура хеш-таблицы
struct hash_table
{
    struct bin* bins;
    unsigned int size;
};
```

Здесь будем считать, что, если поле is_free равно 1, то корзина свободна, если 0 — занята. Соответственно при создании таблицы нам надо будет указать, что корзины хеш-таблицы свободны:

```c
// создание хеш-таблицы
struct hash_table* new_table(unsigned int size)
{
    struct hash_table *table = malloc(sizeof *table);
    table->size = size;
    table->bins = malloc(size * sizeof *table->bins);

    // устанавливаем, что все корзины таблицы не заняты
    struct bin *beg = table->bins, *end = beg + size;
    for (struct bin *bin = beg; bin != end; ++bin) {
        bin->is_free = 1;
    }
    return table;
}
```

Плюс дополнительно мы будем использовать две вспомогательные функции для получения индекса корзины и самой корзины в хеш-таблице:

```c
// получение индекса корзины в хеш-таблице
unsigned int hash_bin_index(struct hash_table *table, unsigned int key)
{
    unsigned int mask = table->size - 1;
    unsigned int index = key & mask;
    return index;
}

// получение указателя на корзину из хеш-таблицы table по индексу key
struct bin* hash_bin(struct hash_table *table, unsigned int key)
{
    return table->bins + hash_bin_index(table, key);
}
```

Теперь рассмотрим базовые операции. Причем пока для упрощения мы будем исходить из того, что коллизии (совпадение хеш-ключей для разных корзин) маловероятны.

## Добавление

При добавлении элемента в хеш-таблицу в качестве примера мы можем получить корзину таблицы и поместить туда ключ (а также другие данные):

```c
void insert_key(struct hash_table *table, unsigned int key)
{
    // получаем указатель на корзину в хеш-таблице
    struct bin *bin = hash_bin(table, key);

    // устанавливаем ключ, если корзина свободна
    if (bin->is_free)
    {
        bin->key = key;
        bin->is_free = 0;    // корзина занята
    }
    // случай else пока не обрабатываем, ограничемся просто выводом диагностического сообщения
    else
    {
        puts("Error. The bin is not empty");
    }
}
```

В данном случае если корзина свободна, помещаем в нее ключ и указываем, что теперь она занята. В данном случае мы пока никак не обрабатываем коллизии, когда на одну корзину может приходиться несколько разных данных.

## Проверка наличия значения

Для проверки наличия ключа в таблице, используем тот же принцип: берем контейнер и проверяем, есть ли там ключ:

```c
int contains_key(struct hash_table *table, unsigned int key)
{
    struct bin* bin = hash_bin(table, key);
    // корзина содержит данные, если она не пуста
    // и содержит именно тот ключ, который мы ищем
    return !bin->is_free && (bin->key == key);
}
```

Здесь мы проверяем, свободен ли контейнер, прежде чем проверять ключ. Это позволит избежать случайной ошибки, когда случайные данные эквивалентны значению ключа.

## Удаление

При удалении в принципе достаточно устанавливать, что корзина свободна, если она содержит искомый ключ:

```c
void delete_key(struct hash_table *table, unsigned int key)
{
    struct bin* bin = hash_bin(table, key);
    bin->is_free = 1;
}
```

Но опять же, в данном случае мы исходим из того, что коллизии маловероятны.

## Полный пример

Для формирования общей картины соберем весь код и в качестве примера проделаем операции над произвольной строкой:

```c
#include <stdio.h>
#include <stdlib.h>

// структура корзины в хеш-таблице
struct bin
{
    int is_free;        // свободная ли корзина
    unsigned int key;   // хеш-ключ
};

// структура хеш-таблицы
struct hash_table
{
    struct bin* bins;
    unsigned int size;
};

// создание хеш-таблицы
struct hash_table* new_table(unsigned int size);
// удаление хеш-таблицы
void delete_table(struct hash_table *table);

// получаем хеш-ключ из строки
unsigned int hash_fnv1a(const char *str);

// получение индекса корзины в хеш-таблице
unsigned int hash_bin_index(struct hash_table *table, unsigned int key);
// получение указателя на корзину из хеш-таблицы table по индексу key
struct bin* hash_bin(struct hash_table *table, unsigned int key);

// добавление данных в хеш-таблицу
void insert_key(struct hash_table *table, unsigned int key);
// проверка наличия данных
int contains_key(struct hash_table *table, unsigned int key);
// удаление
void delete_key(struct hash_table *table, unsigned int key);

int main(void)
{
    // строка для работы
    char* red = "red";
    // получаем хеш строки
    unsigned int red_hash = hash_fnv1a(red);

    // создаем хеш-таблицу
    const unsigned int table_size = 1024;
    struct hash_table* table = new_table(table_size);

    // добавляем строку в хеш-таблицу
    insert_key(table, red_hash);

    // проверяем, есть ли строка в хеш-таблице
    int is_present = contains_key(table, red_hash);
    puts(is_present? "string is present" : "string is absent");

    // удаляем строку из хеш-таблицы
    delete_key(table, red_hash);

    // снова проверяем, есть ли строка в хеш-таблице
    is_present = contains_key(table, red_hash);
    puts(is_present? "string is present" : "string is absent");

    // удаляем таблицу
    delete_table(table);

    return 0;
}

// получаем хеш-ключ из строки
unsigned int hash_fnv1a(const char *str)
{
   unsigned int hash = 2166136261u;

    while (*str) {
        hash ^= (unsigned char)(*str++);
        hash *= 16777619;
    }

    return hash;
}

// создание хеш-таблицы
struct hash_table* new_table(unsigned int size)
{
    struct hash_table *table = malloc(sizeof *table);
    table->size = size;
    table->bins = malloc(size * sizeof *table->bins);

    // устанавливаем, что все корзины таблицы не заняты
    struct bin *beg = table->bins, *end = beg + size;
    for (struct bin *bin = beg; bin != end; ++bin) {
        bin->is_free = 1;
    }
    return table;
}

// удаление хеш-таблицы
void delete_table(struct hash_table *table)
{
    free(table->bins);
    free(table);
}

// получение индекса корзины в хеш-таблице
unsigned int hash_bin_index(struct hash_table *table, unsigned int key)
{
    unsigned int mask = table->size - 1;
    unsigned int index = key & mask;
    return index;
}

// получение указателя на корзину из хеш-таблицы table по индексу key
struct bin* hash_bin(struct hash_table *table, unsigned int key)
{
    return table->bins + hash_bin_index(table, key);
}

// добавление данных в хеш-таблицу
void insert_key(struct hash_table *table, unsigned int key)
{
    struct bin *bin = hash_bin(table, key);

    if (bin->is_free)
    {
        bin->key = key;
        bin->is_free = 0;
    }
    else
    {
        puts("Error. The bin is not empty");
    }
}

// проверка наличия данных
int contains_key(struct hash_table *table, unsigned int key)
{
    struct bin* bin = hash_bin(table, key);
    return !bin->is_free;
}

// удаление
void delete_key(struct hash_table *table, unsigned int key)
{
    struct bin* bin = hash_bin(table, key);
    bin->is_free = 1;
}
```

## Упрощенный альтернативный вариант

В качестве альтернативы возьмем вариант, когда все ячейки хеш-таблицы хранят только числовые хеши данных — своего рода упрощенный вариант. Опять же в данном случае мы будем исходить из идеальной ситуации, когда у нас коллизии/совпадения хешей или индексов корзин для разных значений маловероятны:

```c
#include <stdio.h>
#include <stdlib.h>

// индикатор, что корзина пуста
#define EMPTY 0

// структура хеш-таблицы
struct hash_table
{
    unsigned int *bins;
    unsigned int size;
};

// создание хеш-таблицы
struct hash_table* new_table(unsigned int size);
// удаление хеш-таблицы
void delete_table(struct hash_table *table);

// получаем хеш-ключ из строки
unsigned int hash_fnv1a(const char *str);

// получение индекса корзины в хеш-таблице
unsigned int hash_bin_index(struct hash_table *table, unsigned int key);
// получение указателя на корзину из хеш-таблицы table по индексу key
unsigned int* hash_bin(struct hash_table *table, unsigned int key);

// добавление данных в хеш-таблицу
void insert_key(struct hash_table *table, unsigned int key);
// проверка наличия данных
int contains_key(struct hash_table *table, unsigned int key);
// удаление
void delete_key(struct hash_table *table, unsigned int key);

int main(void)
{
    char* red = "red";
    unsigned int red_hash = hash_fnv1a(red);

    const unsigned int table_size = 1024;
    struct hash_table* table = new_table(table_size);

    insert_key(table, red_hash);

    int is_present = contains_key(table, red_hash);
    puts(is_present? "string is present" : "string is absent");

    delete_key(table, red_hash);

    is_present = contains_key(table, red_hash);
    puts(is_present? "string is present" : "string is absent");

    delete_table(table);

    return 0;
}

// получаем хеш-ключ из строки
unsigned int hash_fnv1a(const char *str)
{
   unsigned int hash = 2166136261u;

    while (*str) {
        hash ^= (unsigned char)(*str++);
        hash *= 16777619;
    }

    return hash;
}

// создание хеш-таблицы
struct hash_table* new_table(unsigned int size)
{
    struct hash_table *table = malloc(sizeof *table);
    table->size = size;
    table->bins = malloc(size * sizeof *table->bins);

    unsigned int *beg = table->bins, *end = beg + size;
    for (unsigned int* bin = beg; bin != end; ++bin) {
        *bin = EMPTY;
    }
    return table;
}

// удаление хеш-таблицы
void delete_table(struct hash_table *table)
{
    free(table->bins);
    free(table);
}

// получение индекса корзины в хеш-таблице
unsigned int hash_bin_index(struct hash_table *table, unsigned int key)
{
    unsigned int mask = table->size - 1;
    unsigned int index = key & mask;
    return index;
}

// получение указателя на корзину из хеш-таблицы table по индексу key
unsigned int* hash_bin(struct hash_table *table, unsigned int key)
{
    return table->bins + hash_bin_index(table, key);
}

// добавление данных в хеш-таблицу
void insert_key(struct hash_table *table, unsigned int key)
{
    if (key == EMPTY)
    {
        puts("Error. The key is invalid");
        return;
    }
    unsigned int* bin = hash_bin(table, key);

    if (*bin == EMPTY)
    {
        *bin = key;
    }
    else
    {
        puts("Error. The bin is not empty");
    }
}

// проверка наличия данных
int contains_key(struct hash_table *table, unsigned int key)
{
    unsigned int* bin = hash_bin(table, key);
    return *bin == key;
}

// удаление
void delete_key(struct hash_table *table, unsigned int key)
{
    unsigned int* bin = hash_bin(table, key);
    *bin = EMPTY;
}
```

Здесь корзины хеш-таблицы представлены указателем на значений unsigned int. Для указания, что корзина пуста используется маркер EMPTY (по сути число 0).

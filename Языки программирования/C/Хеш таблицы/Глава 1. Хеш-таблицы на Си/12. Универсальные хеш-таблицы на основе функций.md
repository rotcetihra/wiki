В прошлой теме было рассмотрено создание универсальных хеш-таблиц с помощью макросов. Однако написание макросов для генерации кода и использование представляет непростую задачу, которая чревата ошибками, а написанный код труден в отладке. Кроме того, макросы могут приводить к раздуванию бинарного кода. Но есть альтернативный подход к написанию универсальных хеш-таблиц - использование единственного универсального типа, который есть в языке C -указатель типа **void**. И именно этот подход мы и рассмотрим в данной статье.

Стандарт языка гарантирует, что переменной типа `void*` можно присвоить указатель любого типа, и что любой переменной типа указатель можно присвоить указатель типа `void*`. Это означает, что мы можем писать код с использованием объекта типа `void*`, и этот код может обрабатывать любые другие типы указателей.

Недостатком этого подхода является то, что здесь нет проверки типов по умолчанию, то есть вы можете присвоить значение типа `int*` переменной типа `void*`, а затем ее значение переменной типа `char*` и никаких проблем с типами не возникнет, однако во время выполнения можно столкнуться со значительными проблемами.

Другой недостаток - такой подход работает только с указателями. Допустим, вам нужно преобразовать строки в целые числа. Строки уже являются указателями, `char*`, поэтому с ними проблем нет, но любое целое число необходимо преобразовать в `int*`. Если у вас есть переменная `i`, вы можете использовать `&i`, но для литерала, например, 26, вам понадобятся уловки, такие как `&(int){26}`:

```c
void* p_v = &(int){26};
int* p_i = p_v;
printf("i: %d\n", *p_i); // i: 26
```

Но в обоих случаях необходимо убедиться, что вставляемое в таблицу значение не изменяется, иначе механизм хеширования не сможет найти его снова. Это означает, что вы не можете снова присвоить значение переменной i, если указатель вставлен без изменений, и, вероятно, вы не можете снова создать выражение &(int){literal}, поскольку компиляторы, как правило, повторно используют свою память. Поэтому распространенное решение состоит в том, что копировать непосредственно в хеш-таблицу хранимые значения, чтобы хеш-таблица была их владельцем.

Поскольку обычно невозможно знать, как копировать, сравнивать и освобождать ключи и значения, необходимо предоставить пользователю механизм, который позволяет ему самому настроить подобный функционал. Суть механизма состоит в том, что передавать указатели на функции, которые реализуют необходимые операции:

```c
// тип для функции хеширования
typedef unsigned int (*hash_func)(void const *);
// тип для функции сравнения
typedef int (*compare_func)(void const *, void const *);
// тип для функции освобождения памяти
typedef void (*destructor_func)(void *);
// тип для функции копирования данных
typedef void *(*copy_func)(void const *);
```

Далее мы можем определить структуры для типов ключей и значений, которые принимают эти указатели:

```c
// тип ключей
struct key_type 
{
    hash_func hash;
    compare_func cmp;
    copy_func cpy;
    destructor_func del;
};
// тип значений
struct value_type 
{
    copy_func cpy;
    destructor_func del;
};
```

То есть нам надо уметь вычислять хеш-значение и сравнивать два ключа, а также уметь копировать и освобождать данные как для ключей, так и для значений.

Далее определим прототипы для базовых операций с хеш-таблицей:

```c
// создание хеш-таблицы
struct hash_table* new_table(struct key_type const *key_type, struct value_type const *value_type);
// удаление хеш-таблицы
void delete_table(struct hash_table *table);
// добавление ключа и значения
void add(struct hash_table *table, void const *key, void const *value);
// удаление данных
void delete_key(struct hash_table *table, void const *key);
// поиск данных
void *const lookup_key(struct hash_table *table, void const *key);
```

Объявления `const` указывают на то, что мы не будем присваивать себе права собственности на ключи и значения.

### Определение хеш-таблицы

В качестве стратегии реализации хеш-таблицы возьмем стратегию линейного зондирования. Структуры ячейки и самой хеш-таблицы будут выглядеть следующим образом:

```c
// структура ячейки хеш-таблицы
struct bin 
{
    int in_probe : 1;       // является ли ячейка частью последовательности зондирования
    int is_empty : 1;       // является ли ячейка пустой
    unsigned int hash_key;  // хеш-ключ
    void *key;              // указатель на ключ
    void *val;              // указатель на значение
};
// структура хеш-таблицы
struct hash_table 
{
    struct bin *bins;
    unsigned int size;
    unsigned int count;
    unsigned int active;
    struct key_type const *key_type;
    struct value_type const *value_type;
};
```

В принципе эти структуры во многом похожи на те, что использовались в прошлых статьях при рассмотрении стратегии линейного зондирования. За тем исключением, что структура ячейки также содержит указатели на ключ и значение, которые надо сохранить, а структура хеш-таблицы будет содержать указатель на определения функций key_type и value_type для управления ключами и значениями в ячейках.

Поскольку мы будем использовать линейное зондирование, то определим следующую функцию зондирования:

```c
// линейное зондирование
unsigned int p(unsigned int key, unsigned int i, unsigned int m)
{
    return (key + i) & (m - 1);
}
```

### Вспомогательные функции

Для упрощения вызова операций добавим несколько вспомогательных функций. Прежде всего определим хеш-функцию, которая использует хеш-функцию типа ключа для сопоставления пользовательского ключа с хеш-ключом:

```c
unsigned int hash(struct hash_table *table, void const *key)
{
    return table->key_type->hash(key);
}
```

Для копирования ключей и значений (key и value) опредлим соответственно функции copy_key и copy_val:

```c
// копирование ключей
void* copy_key(struct hash_table *table, void const *key)
{
    return table->key_type->cpy(key);
}
// копирование значений
void* copy_val(struct hash_table *table, void const *val)
{
    return table->value_type->cpy(val);
}
```

Аналогично определим функции free_key и free_val для освобождения памяти ключей и значений:

```c
void free_key(struct hash_table *table, void *key)
{
    table->key_type->del(key);
}
void free_val(struct hash_table *table, void *val)
{
    table->value_type->del(val);
}
```

И последняя вспомогательная функция - is_active_bin будет проверять, активна ли ячейка таблицы в данный момент (является ли она частью последовательности зондирования и при этом не является пустой):

```c
int is_active_bin(struct bin *bin)
{
    return bin->in_probe && !bin->is_empty;
}
```

### Создание таблицы и изменение размера

Для инициализации таблиц и их обновления при изменении размера определим функцию `init_table()`:

```c
// вспомогательная функция для добавления данных
void add_internal(struct hash_table *table, unsigned int hash_key, void *key_copy, void *value_copy);
// Инициализирует таблицу и затем копирует ячейки из begin в end
void init_table(struct hash_table *table, unsigned int size, struct bin *begin, struct bin *end)
{
    table->bins = malloc(size * sizeof *table->bins);
    table->size = size;
    table->count = 0;
    table->active = 0;
    struct bin empty_bin = {.in_probe = 0, .is_empty = 1};
    for (unsigned int i = 0; i < table->size; i++) 
    {
        table->bins[i] = empty_bin;
    }
    for (struct bin *bin = begin; bin != end; bin++) 
    {
        if (!bin->is_empty) 
            add_internal(table, bin->hash_key, bin->key, bin->val);
    }
}
```

Она получает (новый) размер и (возможно, пустую) последовательность пустых ячеек, а затем обновляет все данные учета и перемещает ячейки в новую таблицу. Функция использует вспомогательную функцию `add_internal()`, которую мы определим позже, когда будем определять весь функционал для добавления данных.

Теперь определим функцию для создания новой таблицы:

```c
#define MIN_SIZE 8
struct hash_table* new_table(struct key_type const *key_type, struct value_type const *value_type)
{
    struct hash_table *table = malloc(sizeof *table);
    table->key_type = key_type;
    table->value_type = value_type;
    init_table(table, MIN_SIZE, NULL, NULL);
    return table;
}
```

И определим функцию для изменения размера таблицы:

```c
void resize(struct hash_table* table, unsigned int new_size)
{
    struct bin *old_bins_begin = table->bins,
                *old_bins_end = old_bins_begin + table->size;
    init_table(table, new_size, old_bins_begin, old_bins_end);
    free(old_bins_begin);
}
```

Для изменения размера вызывается функция `init_table()` с новым размером и существующими ячейками. Поскольку `init_table` перезапишет указатель на ячейки в структуре таблицы, сохраняем указатель на старые ячейки в переменную old_bins_begin, чтобы потом освободить их.

### Удаление таблицы

При удалении таблицы нам нужно освобождать память как структуры хеш-таблицы и ее ячеек, так и данных, которые копируются в таблицу:

```c
void free_bin(struct hash_table* table, struct bin* bin)
{
    if (is_active_bin(bin)) 
    {
        free_key(table, bin->key);
        free_val(table, bin->val);
        bin->is_empty = 1;
        table->active--;
    }
}
void delete_table(struct hash_table *table)
{
    for (struct bin *bin = table->bins; bin != table->bins + table->size; ++bin) 
    {
        free_bin(table, bin);
    }
    free(table->bins);
    free(table);
}
```

Процесс удаления довольно прост: мы проходим по каждой ячейке и освобождаем все ее данные. Мы можем определить, содержит ли ячейка данные, по установленным в ней флагам, и проверить это с помощью вспомогательной функции `is_active_bin()`.

### Поиск

Для проверки наличия ключа в ячейке сначала определим вспомогательную функцию:

```c
int key_in_bin(struct hash_table* table, struct bin* bin, unsigned int hash_key, void const* key)
{
    return is_active_bin(bin) && bin->hash_key == hash_key && table->key_type->cmp(bin->key, key);
}
```

Прежде всего для поиска ключей используем только активные ячейки, то есть не пустые ячейки в последовательности зондирования. Кроме того, функция сравнения типа ключа (table->key_type->cmp) должна подтвердить равенство ключей.

Далее нам нужен код для поиска ячейки, которая содержит ключ:

```c
struct bin* find_key(struct hash_table* table, unsigned int hash_key, void const *key)
{
    for (unsigned int i = 0; i < table->size; i++) 
    {
        struct bin* bin = table->bins + p(hash_key, i, table->size);
        if (!bin->in_probe || key_in_bin(table, bin, hash_key, key)) return bin;
    }
}
void *const lookup_key(struct hash_table *table, void const *key)
{
    struct bin *bin = find_key(table, hash(table, key), key);
    return bin->in_probe ? bin->val : NULL;
}
```

Эта функция возвращает указатель на значение, хранящееся в таблице, поэтому это не копия.

### Удаление элементов

При удалении используем вышеопределенную функцию find_key() для поиска ячейки с нужным хеш-ключом и затем удаляем ее:

```c
void delete_key(struct hash_table* table, void const *key)
{
    struct bin *bin = find_key(table, hash(table, key), key);
    free_bin(table, bin);
    if (table->active < table->size / 8 && table->size > MIN_SIZE)
        resize(table, table->size / 2);
}
```

### Поиск ячеек

Кроме поиска элементов в хеш-таблице нам также необходим функционал для поиска ячеек:

```c
struct bin* find_empty(struct hash_table *table, unsigned int hash_key)
{
    for (unsigned int i = 0; i < table->size; i++) 
    {
        struct bin *bin = table->bins + p(hash_key, i, table->size);
        if (bin->is_empty) return bin;
    }
}

struct bin* get_bin(struct hash_table* table, unsigned int hash_key, void *const key)
{
    struct bin *bin = find_key(table, hash_key, key);
    return bin->in_probe ? bin : find_empty(table, hash_key);
}
```

Первая функция - `find_empty()` возвращает пустую ячейку. Вторая функция - `get_bin()` призвана найти ячейку с определенным ключом. Однако если нужного ключа не оказалось, то функция возвращает первую, попавшуюся свободную ячейку.

### Добавление элементов

Функционал добавления будет состоять из трех функций. Первая функция будет отвечать за сохранение пары ключ/значение в ячейке:

```c
void store_in_bin(struct hash_table* table, struct bin *bin, unsigned int hash_key, void *key, void *value)
{
    table->active += bin->is_empty;
    table->count += !bin->in_probe;
    free_bin(table, bin);
    *bin = (struct bin) { .in_probe = 1, .is_empty = 0, .hash_key = hash_key, .key = key, .val = value };
}
```

Далее определим вспомогательную функцию для добавления - `add_internal()`:

```c
void add_internal(struct hash_table* table, unsigned int hash_key, void* key_copy, void* value_copy)
{
    struct bin *bin = get_bin(table, hash_key, key_copy);
    store_in_bin(table, bin, hash_key, key_copy, value_copy);
    if (table->count > table->size / 2) resize(table, table->size * 2);
}
```

И основная функция добавления:

```c
void add(struct hash_table* table, void const *key, void const *value)
{
    unsigned int hash_key = hash(table, key);
    void *key_copy = copy_key(table, key);
    void *value_copy = copy_val(table, value);
    add_internal(table, hash_key, key_copy, value_copy);
}
```

### Полный пример применения

Рассмотрим на примере, как мы можем применить весь выше описанный функционал хеш-таблицы. В качестве используемого типа данных возьмем строки, причем и ключи, и значения будут строками:

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef unsigned int (*hash_func)(void const *);
typedef int (*compare_func)(void const *, void const *);
typedef void (*destructor_func)(void *);
typedef void *(*copy_func)(void const *);

struct key_type 
{
    hash_func hash;
    compare_func cmp;
    copy_func cpy;
    destructor_func del;
};
struct value_type 
{
    copy_func cpy;
    destructor_func del;
};

struct bin 
{
    int in_probe : 1;
    int is_empty : 1;
    unsigned int hash_key;
    void *key;
    void *val;
};
struct hash_table 
{
    struct bin *bins;
    unsigned int size;
    unsigned int count;
    unsigned int active;
    struct key_type const *key_type;
    struct value_type const *value_type;
};

unsigned int p(unsigned int key, unsigned int i, unsigned int m)
{
    return (key + i) & (m - 1);
}

unsigned int hash(struct hash_table *table, void const *key)
{
    return table->key_type->hash(key);
}

void* copy_key(struct hash_table *table, void const *key)
{
    return table->key_type->cpy(key);
}
void* copy_val(struct hash_table *table, void const *val)
{
    return table->value_type->cpy(val);
}
void free_key(struct hash_table *table, void *key)
{
    table->key_type->del(key);
}
void free_val(struct hash_table *table, void *val)
{
    table->value_type->del(val);
}
int is_active_bin(struct bin *bin)
{
    return bin->in_probe && !bin->is_empty;
}

void add_internal(struct hash_table *table, unsigned int hash_key, void *key_copy, void *value_copy);
void init_table(struct hash_table *table, unsigned int size, struct bin *begin, struct bin *end)
{
    table->bins = malloc(size * sizeof *table->bins);
    table->size = size;
    table->count = 0;
    table->active = 0;
    struct bin empty_bin = {.in_probe = 0, .is_empty = 1};
    for (unsigned int i = 0; i < table->size; i++) 
    {
        table->bins[i] = empty_bin;
    }
    for (struct bin *bin = begin; bin != end; bin++) 
    {
        if (!bin->is_empty) 
            add_internal(table, bin->hash_key, bin->key, bin->val);
    }
}

#define MIN_SIZE 8
struct hash_table* new_table(struct key_type const *key_type, struct value_type const *value_type)
{
    struct hash_table *table = malloc(sizeof *table);
    table->key_type = key_type;
    table->value_type = value_type;
    init_table(table, MIN_SIZE, NULL, NULL);
    return table;
}

void resize(struct hash_table* table, unsigned int new_size)
{
    struct bin *old_bins_begin = table->bins,
                *old_bins_end = old_bins_begin + table->size;
    init_table(table, new_size, old_bins_begin, old_bins_end);
    free(old_bins_begin);
}

void free_bin(struct hash_table* table, struct bin* bin)
{
    if (is_active_bin(bin)) 
    {
        free_key(table, bin->key);
        free_val(table, bin->val);
        bin->is_empty = 1;
        table->active--;
    }
}
void delete_table(struct hash_table *table)
{
    for (struct bin *bin = table->bins; bin != table->bins + table->size; ++bin) 
    {
        free_bin(table, bin);
    }
    free(table->bins);
    free(table);
}

int key_in_bin(struct hash_table* table, struct bin* bin, unsigned int hash_key, void const* key)
{
    return is_active_bin(bin) && bin->hash_key == hash_key && table->key_type->cmp(bin->key, key);
}

struct bin* find_key(struct hash_table* table, unsigned int hash_key, void const *key)
{
    for (unsigned int i = 0; i < table->size; i++) 
    {
        struct bin* bin = table->bins + p(hash_key, i, table->size);
        if (!bin->in_probe || key_in_bin(table, bin, hash_key, key)) return bin;
    }
}
void *const lookup_key(struct hash_table *table, void const *key)
{
    struct bin *bin = find_key(table, hash(table, key), key);
    return bin->in_probe ? bin->val : NULL;
}

void delete_key(struct hash_table* table, void const *key)
{
    struct bin *bin = find_key(table, hash(table, key), key);
    free_bin(table, bin);
    if (table->active < table->size / 8 && table->size > MIN_SIZE)
        resize(table, table->size / 2);
}

struct bin* find_empty(struct hash_table *table, unsigned int hash_key)
{
    for (unsigned int i = 0; i < table->size; i++) 
    {
        struct bin *bin = table->bins + p(hash_key, i, table->size);
        if (bin->is_empty) return bin;
    }
}

struct bin* get_bin(struct hash_table* table, unsigned int hash_key, void *const key)
{
    struct bin *bin = find_key(table, hash_key, key);
    return bin->in_probe ? bin : find_empty(table, hash_key);
}

void store_in_bin(struct hash_table* table, struct bin *bin, unsigned int hash_key, void *key, void *value)
{
    table->active += !!bin->is_empty;
    table->count += !bin->in_probe;
    free_bin(table, bin);
    *bin = (struct bin) { .in_probe = 1, .is_empty = 0, .hash_key = hash_key, .key = key, .val = value };
}
void add_internal(struct hash_table* table, unsigned int hash_key, void* key_copy, void* value_copy)
{
    struct bin *bin = get_bin(table, hash_key, key_copy);
    store_in_bin(table, bin, hash_key, key_copy, value_copy);
    if (table->count > table->size / 2) resize(table, table->size * 2);
}
void add(struct hash_table* table, void const *key, void const *value)
{
    unsigned int hash_key = hash(table, key);
    void *key_copy = copy_key(table, key);
    void *value_copy = copy_val(table, value);
    add_internal(table, hash_key, key_copy, value_copy);
}

unsigned int hash_fnv1a(const char *str) 
{
   unsigned int hash = 2166136261u;
    while (*str) {
        hash ^= (unsigned char)(*str++);
        hash *= 16777619;
    }
    return hash;
}

{
    return hash_fnv1a((char const*) key);
```

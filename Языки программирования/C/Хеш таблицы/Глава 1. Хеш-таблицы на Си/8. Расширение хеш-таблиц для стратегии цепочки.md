Несмотря на то что метод цепочек позволяет добавлять бесконечное количество элементов в одну корзину, используя связанные списки, расширение хеш-таблицы позволяет увеличить ее главного преимущество - скорость. Смысл в подобном расширении заключается в поддержании константного времени доступа `O(1)`. В частности, следует учитывать следующие аспекты:

-   Производительность: если количество элементов (`n`) значительно превышает количество корзин (`m`), средняя длина списков растет. В итоге поиск превращается в линейный перебор списка, и скорость падает с `O(1)` до `O(n)`.
-   Коэффициент заполнения (Load Factor): для цепочек оптимальным считается `alpha = n/m` в пределах `0.75–1.0`. Когда это значение превышается, это может быть сигналом к увеличению таблицы (обычно в 2 раза) и проводить перехеширование (rehashing), чтобы распределить элементы по новым корзинам и снова укоротить списки.
-   Худший случай: без расширения даже при хорошей хеш-функции мы рискуем получить "вырожденную" таблицу, которая по сути станет одним длинным связным списком.

То есть в данном случае расширение нужно не для того, чтобы новые элементы влезли в хеш-таблицу (как в методе открытой адресации), а для того, чтобы все работало быстро.

#### Аспекты реализации

В качестве отправной точки возьмем код работы с хеш-таблицами по методу цепочки и рассмотрим те моменты, которые будут отличаться. А в конце я приведу общий код с простейшим примером работы.

Во-первых, необходимо отслеживать количество корзин и количество ключей, которые хранятся в таблице. Для этого в структуре хеш-таблицы определим два поля: количество корзин и количество элементов, которые хранятся в этих корзинах:

```c
// отдельный элемент корзины
struct node 
{
    unsigned int key;
    struct node* next;
};
// определение корзины
typedef struct node **BIN;

// структура хеш-таблицы
struct hash_table 
{
    struct node** bins;
    unsigned int size;  // общее количество корзин
    unsigned int count;  // количество добавленных элементов (НЕ корзин!)
};
```

При создании таблицы инициализируем ее минимальным количеством ячеек (MIN_SIZE),и сохраняем это значение в ее размере:

```c
#define MIN_SIZE 8

void init_bins(struct hash_table* table)
{
    // для всех ячеек устанавливаем в качестве значения устанавливаем NULL
    for (BIN bin = table->bins; bin < table->bins + table->size; bin++) 
    {
        *bin = NULL;
    }
}
struct hash_table* new_table()
{
    struct hash_table *table = malloc(sizeof *table);
    struct node** bins = malloc(MIN_SIZE * sizeof *bins);
    *table = (struct hash_table){.bins = bins, .size = MIN_SIZE, .count = 0};
    init_bins(table);   // инициализация всех корзин таблицы
    return table;
}
```

При изменении размера необходимо выделить новый массив для ячеек, а затем скопировать все данные из старых ячеек в новые. Этот процесс можно разделить на две функции. Первая функция выделяет новые ячейки и вызывает вторую для перемещения данных:

```c
// копирование корзин и их узлов, from - указатель на первую корзину, to - указатель на последнюю
void copy_nodes(struct hash_table* table, BIN from, BIN to)
{
    // проходим от from до to
    for (; from < to; from++) 
    {
        // пока не дойдем до указателя на node, который равен NULL
        while (*from) 
        {
            struct node* node = *from;
            // удаляем первый узел из старой корзины и заменяем его на next
            *from = node->next;
            // помещаем узел в новую корзину
            BIN new_bin = get_bin(table, node->key);
            node->next = *new_bin;
            *new_bin = node;
        }
    }
}
// изменение размера
void resize(struct hash_table* table, unsigned int new_size)
{

    struct node **old_bins = table->bins,        // получаем старые корзины
                **old_from = old_bins,              // начальная позиция для копирования
                **old_to = old_from + table->size;  // конечная позиция для копирования
    // выделяем память под новую таблицу
    table->bins = malloc(new_size * sizeof *table->bins);
    table->size = new_size;
    init_bins(table);
    // копируем данные из старых корзин в новые
    copy_nodes(table, old_from, old_to);
    // освобождаем память корзин старой хеш-таблицы
    free(old_bins);
}
```

Функция copy_nodes принимает хеш-таблицу, указатель на первую корзину - from (указатель на указатель на node) и указатель на последнюю корзину, и копирует все элементы из старой корзины в новую. А функция resize получает копируемую хеш-таблицу и новый размер, получает старые корзины и, используя функцию copy_nodes, копирует данные в новые корзины.

Изменять размер можно, например, каждый раз при вставке или удалении ключа:

```c
// вставка ключа
void insert_key(struct hash_table* table, unsigned int key)
{
    // получаем корзину
    BIN bin = get_bin(table, key);
    if (!contains_node(bin, key)) 
    {
        add_node(bin, key);
        table->count++;
        // если в таблице количество элементов равно количеству корзин
        // то изменяем размер
        if (table->size == table->count) 
        {
            resize(table, 2 * table->size);
        }
    }
}
// удаление ключа 
void delete_key(struct hash_table* table, unsigned int key)
{
    BIN bin = get_bin(table, key);
    if (contains_node(bin, key)) 
    {
        free_node(bin, key);
        table->count--;
        // если в таблице количество элементов - четверть от количества корзин
        // то изменяем размер
        if (table->size > MIN_SIZE && table->count < table->size / 4) 
        {
            resize(table, table->size / 2);
        }
    }
}
```

Перед вставкой/удалением ключа сначала необходимо проверить, находится ли нужный ключ в таблице. И если да, то изменение размера происходит после вставки или удаления. ПРи вставке расширяем таблицу, если в таблице количество элементов равно количеству корзин. При удалении сужаем таблицу, если в таблице количество элементов - четверть от количества корзин.

#### Полный пример

Полный пример кода для работы с хеш-таблицей с учетом ее расширения мог бы выглядеть следующим образом:

```c
#include <stdio.h>
#include <stdlib.h>

// минимальное количество корзин
#define MIN_SIZE 8

// отдельный элемент корзины
struct node 
{
    unsigned int key;
    struct node* next;
};
// определение корзины
typedef struct node** BIN;

// структура хеш-таблицы
struct hash_table 
{
    struct node** bins;
    unsigned int size;  // общее количество корзин
    unsigned int count;  // количество добавленных элементов (НЕ корзин!)
};

// создаем новый узел в корзине
struct node* new_node(unsigned int key, struct node *next)
{
    struct node *node = malloc(sizeof *node);
    *node = (struct node){.key = key, .next = next};
    return node;
}
// добавляем новый узел в корзину
void add_node(BIN bin, unsigned int key)
{
    *bin = new_node(key, *bin);
}

// поиск ключа в корзине
BIN find_key(BIN bin, unsigned int key)
{
    for (; *bin; bin = &(*bin)->next) 
    {
        if ((*bin)->key == key) return bin;
    }
    return NULL;
}
// проверка наличия элемента
int contains_node(BIN bin, unsigned int key)
{
    return find_key(bin, key) != NULL;
}
// удаляем начало (голову) корзины
void free_head(BIN bin)
{
    struct node *next = (*bin)->next;
    free(*bin);
    *bin = next;
}

// удаление элемента
void free_node(BIN bin, unsigned int key)
{
    if ((bin = find_key(bin, key))) 
    {
        free_head(bin);
    }
}

// удаляем всю корзину
void free_bin(BIN bin)
{
    while (*bin) 
    {
        free_head(bin);
    }
}

// инициализация корзин
void init_bins(struct hash_table* table)
{
    // для всех ячеек устанавливаем в качестве значения устанавливаем NULL
    for (BIN bin = table->bins; bin < table->bins + table->size; bin++) 
    {
        *bin = NULL;
    }
}
// создание новой таблицы
struct hash_table* new_table()
{
    struct hash_table *table = malloc(sizeof *table);
    struct node** bins = malloc(MIN_SIZE * sizeof *bins);
    *table = (struct hash_table){.bins = bins, .size = MIN_SIZE, .count = 0};
    init_bins(table);   // инициализация всех корзин таблицы
    return table;
}
// удаление хеш-таблицы
void free_table(struct hash_table *table)
{
    for (BIN bin = table->bins; bin < table->bins + table->size; bin++) 
    {
        free_bin(bin);
    }
    free(table->bins);
    free(table);
}
// получение корзины в хеш-таблице
BIN get_bin(struct hash_table *table, unsigned int key)
{
    unsigned int mask = table->size - 1;
    unsigned int index = key & mask;
    return table->bins + index;
}


void copy_nodes(struct hash_table* table, BIN from, BIN to)
{
    // проходим корзины от from до to
    for (; from < to; from++) 
    {
        // пока не дойдем до указателя на node, который равен NULL
        while (*from) 
        {
            struct node* node = *from;
            // удаляем первый узел из старой корзины и заменяем его на next
            *from = node->next;
            // помещаем узел в новую корзину
            BIN new_bin = get_bin(table, node->key);
            node->next = *new_bin;
            *new_bin = node;
        }
    }
}

void resize(struct hash_table* table, unsigned int new_size)
{

    struct node **old_bins = table->bins,           // получаем старые корзины
                **old_from = old_bins,              // начальная позиция для копирования
                **old_to = old_from + table->size;  // конечная позиция для копирования
    // выделяем память под новую таблицу
    table->bins = malloc(new_size * sizeof *table->bins);
    table->size = new_size;
    init_bins(table);
    // копируем данные из старыз корзин в новые
    copy_nodes(table, old_from, old_to);
    // освобождаем память корзин старой хеш-таблицы
    free(old_bins);
}


// добавление данных в хеш-таблицу
void insert_key(struct hash_table* table, unsigned int key)
{
    // получаем корзину
    BIN bin = get_bin(table, key);
    if (!contains_node(bin, key)) 
    {
        add_node(bin, key);
        table->count++;
        // если в таблице количество элементов равно количеству корзин
        // то изменяем размер
        if (table->size == table->count) 
        {
            resize(table, 2 * table->size);
        }
    }
}
// удаление ключа
void delete_key(struct hash_table* table, unsigned int key)
{
    BIN bin = get_bin(table, key);
    if (contains_node(bin, key)) 
    {
        free_node(bin, key);
        table->count--;
        // если в таблице количество элементов - четверть от количества корзин
        // то изменяем размер
        if (table->size > MIN_SIZE && table->count < table->size / 4) 
        {
            resize(table, table->size / 2);
        }
    }
}
// проверка наличия данных
int contains_key(struct hash_table *table, unsigned int key)
{
    return contains_node(get_bin(table, key), key);
}
// получаем хеш-значение из строки
unsigned int hash_fnv1a(const char *str) 
{
   unsigned int hash = 2166136261u;

    while (*str) {
        hash ^= (unsigned char)(*str++);
        hash *= 16777619;
    }

    return hash;
}

// примитивная функция для вывода хеш-таблицы
void print_table(struct hash_table* table)
{
    for (int i =0; i < table->size; ++i) 
    {
        BIN bin = table->bins + i;
        // выводим все ключи текущей корзины
        printf("bin %d: ", i);
        for (; *bin; bin = &(*bin)->next) 
        {
            printf("%u, ", (*bin)->key);
        }

        putchar(10);
    }
}

int main(void)
{
    struct hash_table* dict = new_table();

    // строки для примера
    char* colors[] = {"red", "green", "blue", "yellow", "violet", "rose", "black", "white"};
    for(int i =0; i < sizeof(colors)/sizeof(colors[0]); i++)
    {
        // получаем хеш строк
        unsigned int hash = hash_fnv1a(colors[i]);
        // добавляем строки в хеш-таблицу
        insert_key(dict, hash);
    }
    print_table(dict);
    // получаем хеш строки `red`
    unsigned int red_hash = hash_fnv1a(colors[0]); 

    // проверим, есть ли строка red в таблице
    int is_present = contains_key(dict, red_hash);
    printf("string `red` is %s\n", is_present?  "present" : "absent");

    // удаляем строку `red` из хеш-таблицы
    delete_key(dict, red_hash);

    // снова проверяем, есть ли строка в хеш-таблице
    is_present = contains_key(dict, red_hash);
    printf("string `red` is %s\n", is_present?  "present" : "absent");

    // удаляем всю хеш-таблицу
    free_table(dict);

    return 0;
}
```

В данном случае в качестве примера добавляем все строки массива colors в хеш-таблицу, выводим ее содержимое с помощью простейшей функции print_table и затем проверяем два раза наличие ключа в таблицы - до и после его удаления.

Хеш-функция — это функция, которая преобразует входные данные произвольного размера в фиксированное числовое значение или хеш. В общем случае, условно говоря, мы берем произвольные данные и преобразуем их в некоторое число в диапазоне от 0 до N-1, где N — размер таблицы.

Хорошая хеш-функция обладает несколькими ключевыми свойствами:

- **детерминированность:** одинаковые входные данные всегда дают одинаковый результат
- **равномерность распределения:** хеши должны равномерно распределяться по всему диапазону значений, минимизируя количество коллизий
- **скорость вычисления:** функция должна работать быстро, иначе теряется весь смысл использования хеш-таблицы

В целом в программировании хеш-функции представляют фундаментальный инструмент для построения быстрых структур данных (например, хеш-таблиц), для оптимизации поиска, для проверки целостности и обеспечения безопасности. Различают некриптографические (для хеш-таблиц, индексов, словарей) и криптографические (для безопасности, цифровых подписей, хранения паролей). В языке C хеш-функции активно применяются в структурах данных, криптографии, проверке целостности, кэшировании и компиляторах.

Ключевой компонент хеш-функции — это ее алгоритм, в соответствии с которым вычисляется хеш. И таких алгоритмов может быть бесконечное множество. Для примера рассмотрим сначала простейшие реализации хеш-функций для строк. Так, возьмем простейшую, но неэффективную реализацию в виде суммирования ASCII-кодов символов:

```c
unsigned int hash_simple(const char *key, unsigned int table_size) {
    unsigned int hash = 0;
    while (*key) {
        hash += (unsigned char)(*key);
        key++;
    }
    return hash % table_size;
}
```

Проблема такой функции в том, что строки "abc" и "cab" дадут одинаковый хеш:

```c
#include <stdio.h>

unsigned int hash_simple(const char *key, unsigned int table_size) {
    unsigned int hash = 0;
    while (*key) {
        hash += (unsigned char)(*key);
        key++;
    }
    return hash % table_size;
}

int main(void)
{
    const unsigned table_size = 1024;

    char* abc = "abc";
    char* cab = "cab";

    unsigned abc_hash = hash_simple(abc, table_size);
    unsigned cab_hash = hash_simple(cab, table_size);

    printf("String: %s \thash: %u \n",abc, abc_hash);
    printf("String: %s \thash: %u \n",cab, cab_hash);
    return 0;
}
```

Консольный вывод:

```
String: abc 	hash: 294
String: cab 	hash: 294
```

Значительно лучше работает полиномиальная хеш-функция, которая учитывает позицию каждого символа:

```c
unsigned int hash_polynomial(const char *key, unsigned int table_size) {
    unsigned int hash = 0;
    int prime = 31;
    while (*key) {
        hash = hash * prime + (unsigned char)(*key);
        key++;
    }
    return hash % table_size;
}
```

Ее применение:

```c
#include <stdio.h>

unsigned int hash_polynomial(const char *key, unsigned int table_size) {
    unsigned int hash = 0;
    int prime = 31;
    while (*key) {
        hash = hash * prime + (unsigned char)(*key);
        key++;
    }
    return hash % table_size;
}

int main(void)
{
    const unsigned table_size = 1024;

    char* abc = "abc";
    char* cab = "cab";
    unsigned abc_hash = hash_polynomial(abc, table_size);
    unsigned cab_hash = hash_polynomial(cab, table_size);
    printf("String: %s \thash: %u \n",abc, abc_hash);
    printf("String: %s \thash: %u \n",cab, cab_hash);
    return 0;
}
```

Консольный вывод:

```
String: abc 	hash: 98
String: cab 	hash: 964
```

## DJB2

Одной из популярных простейших и одновременно практических хеш-функций считается djb2, разработанная Дэниелом Бернштейном:

```c
unsigned long hash_djb2(const char *str) {
    unsigned long hash = 5381;
    int c;
    while ((c = (unsigned char)*str++)) {
        hash = ((hash << 5) + hash) + c; /* hash * 33 + c */
    }
    return hash;
}
```

Принцип работы этой хеш-функции:

- Она принимает данные произвольной длины (формально это строка, но фактически это может быть произвольный набор байтов)
- Использует умножение на 33 (эффективно через сдвиг)
- Возвращает число фиксированного размера
- Стремится минимизировать коллизии (одинаковый хеш для разных данных)

Магические числа 5381 и 33 подобраны эмпирически и обеспечивают хорошее распределение битов. В итоге такая хеш-функция работает быстро и является простой в реализации. Пример применения:

```c
#include <stdio.h>

unsigned long hash_djb2(const char *str) {
    unsigned long hash = 5381;
    int c;
    while ((c = (unsigned char)*str++)) {
        hash = ((hash << 5) + hash) + c; /* hash * 33 + c */
    }
    return hash;
}

int main(void)
{
    char* abc = "abc";
    char* cab = "cab";
    unsigned abc_hash = hash_djb2(abc);
    unsigned cab_hash = hash_djb2(cab);
    printf("String: %s \thash: %u \n",abc, abc_hash);
    printf("String: %s \thash: %u \n",cab, cab_hash);
    return 0;
}
```

Результат:

```
String: abc 	hash: 193485963
String: cab 	hash: 193488107
```

## FNV-1a

Для примера рассмотрим еще одну популярную функцию — FNV-1a (Fowler-Noll-Vo, вариант 1a). Этот алгоритм FNV-1a особенно популярен в системном программировании, базах данных и сетевых приложениях, где скорость важнее криптографической безопасности:

```c
#include <stdint.h>

uint32_t hash_fnv1a(const char *str) {
    uint32_t hash = 2166136261u;

    while (*str) {
        hash ^= (unsigned char)(*str++);
        hash *= 16777619;
    }

    return hash;
}
```

Эта функция также принимает некоторую строку и возвращает хеш — 32-разрядное целое число без знака. Вначале определяется переменная hash — это базис смещения FNV-1a (константа, специфичная для 32-битного FNV). Это тщательно подобранное простое число, которое служит начальным значением хеша.

Затем в цикле функция обрабатывает последовательно каждый символ строки. Сначала выполняет операцию XOR с текущим значением хеша и перемещает указатель на следующий символ:

```c
hash ^= (unsigned char)(*str++)
```

Затем умножает хеш на простое число FNV-1a (16777619):

```c
hash *= 16777619
```

Это простое число было выбрано из-за его математических свойств, которые помогают создать хорошее распределение. Ее преимущества: хорошее распределение, простота и высокая скорость.

```c
#include <stdio.h>
#include <stdint.h>

uint32_t hash_fnv1a(const char *str) {
    uint32_t hash = 2166136261u;

    while (*str) {
        hash ^= (unsigned char)(*str++);
        hash *= 16777619;
    }

    return hash;
}

int main(void)
{
    char* abc = "abc";
    char* cab = "cab";
    unsigned abc_hash = hash_fnv1a(abc);
    unsigned cab_hash = hash_fnv1a(cab);
    printf("String: %s \thash: %u \n",abc, abc_hash);
    printf("String: %s \thash: %u \n",cab, cab_hash);
    return 0;
}
```

Результат программы:

```
String: abc 	hash: 440920331
String: cab 	hash: 4101259185
```

Пример пошагового выполнения функции хеширования для строки "abc":

1. Начало выполнения функции: hash = 2166136261
2. Начинаем обрабатывать первый символ — 'a' (или числовое значение 97):
   - XOR: 2166136261 ^ 97 = 2166136260
   - Умножение: 2166136260 * 16777619. В результате получаем некоторое большое число, деленное по модулю на 2^32
3. Обрабатываем второй символ — 'b' (98): XOR с результатом обработки предыдущего символа, затем умножение
4. Обрабатываем третий символ — 'c' (99): XOR с результатом обработки предыдущего символа, затем умножение

Это практичная реализация, часто используемая по следующим причинам:

- Она чрезвычайно быстрая (только XOR и умножение на байт)
- Отсутствие сложных зависимостей или выделения памяти
- Хорошо работает с короткими строками (например, ключами словарей, именами переменных)
- Часто используется в хеш-таблицах, компиляторах и интерпретаторах

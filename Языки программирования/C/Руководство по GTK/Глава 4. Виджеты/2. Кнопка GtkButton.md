Кнопки являются одним из базовых элементов пользовательского интерфейса, позволяя пользователю с помощью нажатий взаимодействовать с приложением. В GTK простые кнопки представлены типом GtkButton. Для создания новой кнопки можно использовать ряд функций:

- `gtk_button_new`: Создает новый виджет GtkButton.
- `gtk_button_new_from_icon_name`: Создает новую кнопку, содержащую значок из текущей темы значков.
- `gtk_button_new_with_label`: Создает виджет GtkButton с дочерним элементом GtkLabel.
- `gtk_button_new_with_mnemonic`: Создает новый виджет GtkButton, содержащий метку.

Используем функцию `gtk_button_new_with_label()`, в которую передается текст, который будет отображаться на кнопке:

```c
#include <gtk/gtk.h>

static void app_activate (GApplication *app, gpointer *user_data) {

    GtkWidget *window = gtk_application_window_new (GTK_APPLICATION (app));
    gtk_window_set_title (GTK_WINDOW (window), "METANIT.COM");
    gtk_window_set_default_size (GTK_WINDOW (window), 250, 200);
    
    // создаем кнопку 
    GtkWidget *button = gtk_button_new_with_label("Click Me");
    
    gtk_window_set_child (GTK_WINDOW (window), button);
    gtk_window_present (GTK_WINDOW (window));
}

int main (int argc, char **argv)
{
    GtkApplication *app = gtk_application_new ("com.metanit", G_APPLICATION_DEFAULT_FLAGS);
    g_signal_connect (app, "activate", G_CALLBACK (app_activate), NULL);
    int status = g_application_run (G_APPLICATION (app), argc, argv);
    g_object_unref(app);
    return status;
}
```

В данном случае создается кнопка, которая содержит текст "Click Me".

По умолчанию кнопка занимает все пространство контейнера - окна.

Определение аналогичной кнопки в XML (файл "builder.ui"):

```xml
<?xml version="1.0" encoding="UTF-8"?>
<interface>
  <object class="GtkWindow" id="window">
    <property name="title">METANIT.COM</property>
    <property name="default-width">300</property>
    <property name="default-height">250</property>
    <child>
      <object class="GtkButton" id="button">
        <property name="label">Click Me</property>
      </object>
    </child>
  </object>
</interface>
```

И основной файл приложения, где загружается интерфейс:

```c
#include <gtk/gtk.h>

static void app_activate (GtkApplication *app, gpointer *user_data) {

    GtkBuilder* builder = gtk_builder_new_from_file("builder.ui");
    GObject* window = gtk_builder_get_object (builder, "window");
    g_object_unref (builder);
    gtk_window_set_application (GTK_WINDOW (window), app);
    gtk_window_present (GTK_WINDOW (window));
}

int main (int argc, char **argv)
{
    GtkApplication *app = gtk_application_new ("com.metanit", G_APPLICATION_DEFAULT_FLAGS);
    g_signal_connect (app, "activate", G_CALLBACK (app_activate), NULL);
    int status = g_application_run (G_APPLICATION (app), argc, argv);
    g_object_unref(app);

    return status;
}
```

Вкратце рассмотрим функционал виджета. Основные свойства:

- child: дочерний виджет
- label: текст кнопки
 
Основные функции:

- gtk_button_get_child: возвращает вложенный виджет
- gtk_button_get_label: возвращает текст кнопки
- gtk_button_set_child: устанавливает вложенный виджет
- gtk_button_set_label: устанавливает текст кнопки

Разберем вкратце основной функционал виджета.

## Обработка нажатия кнопки

Ключевой особенностью кнопок является возможность с помощью нажатий посылать приложению сигналы, а приложение в ответ на нажатия может выполнить некоторые действия. Для обработки нажатия кнопки GtkButton нам надо прикрепить к сигналу "clicked" функцию обработчика:

```c
void clicked (GtkButton* self, gpointer user_data)
```

Первый параметр обработчика - сама нажатая кнопка, а второй - некоторые данные, которые передаются в обработчик. Например, при нажатии на кнопку выведем сообщение на консоль:

```c
#include <gtk/gtk.h>

static void click_button (GtkButton *btn) {
    g_print ("Clicked.\n");
}

static void app_activate (GApplication *app, gpointer *user_data) {

    GtkWidget *window = gtk_application_window_new (GTK_APPLICATION (app));
    gtk_window_set_title (GTK_WINDOW (window), "METANIT.COM");
    gtk_window_set_default_size (GTK_WINDOW (window), 250, 200);
    
    // создаем метку 
    GtkWidget *button = gtk_button_new_with_label("Click Me");
    g_signal_connect (button, "clicked", G_CALLBACK (click_button), NULL);

    gtk_window_set_child (GTK_WINDOW (window), button);
    gtk_window_present (GTK_WINDOW (window));
}

int main (int argc, char **argv)
{
    GtkApplication *app = gtk_application_new ("com.metanit", G_APPLICATION_DEFAULT_FLAGS);
    g_signal_connect (app, "activate", G_CALLBACK (app_activate), NULL);
    int status = g_application_run (G_APPLICATION (app), argc, argv);
    g_object_unref(app);
    return status;
}
```

Как и в случае с активацией/обработкой запуска приложения для прикрепления функции обработчика к сигналу "clicked" применяется функция `g_signal_connect()`:

```c
g_signal_connect(button, "clicked", G_CALLBACK (click_button), NULL);
```

Функция обработчика называется "click_button" и просто выводит строку на консоль. Причем поскольку в данном случае второй параметр сигнала "clicked" нам не важен, функция обработчика принимает только один параметр - саму кнопку:

```c
static void click_button (GtkButton *btn) {
    g_print ("Clicked.\n");
}
```

В итоге при нажатии на кнопку на кониоль будет выведено сообщение.

## Обращение к дочерней метке

Текст на кнопке представлен вложенным элементом GtkLabel. Для получения вложенного содержимого для кнопки определена функция `gtk_button_get_child()`:

```c
GtkWidget* gtk_button_get_child (GtkButton* button)
```

Получив дочерний элемент (метку), мы можем как-то ее изменить - например, изменить текст метки. Так, по нажатию на кнопку изменим текст метки:

```c
#include <gtk/gtk.h>

static void click_button (GtkButton *btn) {

    GtkWidget* label = gtk_button_get_child (btn);
    gtk_label_set_text(GTK_LABEL(label), "Clicked!!!");
}

static void app_activate (GApplication *app, gpointer *user_data) {

    GtkWidget *window = gtk_application_window_new (GTK_APPLICATION (app));
    gtk_window_set_title (GTK_WINDOW (window), "METANIT.COM");
    gtk_window_set_default_size (GTK_WINDOW (window), 250, 200);
    
    // создаем метку 
    GtkWidget *button = gtk_button_new_with_label("Click Me");
    g_signal_connect (button, "clicked", G_CALLBACK (click_button), NULL);

    gtk_window_set_child (GTK_WINDOW (window), button);
    gtk_window_present (GTK_WINDOW (window));
}

int main (int argc, char **argv)
{
    GtkApplication *app = gtk_application_new ("com.metanit", G_APPLICATION_DEFAULT_FLAGS);
    g_signal_connect (app, "activate", G_CALLBACK (app_activate), NULL);
    int status = g_application_run (G_APPLICATION (app), argc, argv);
    g_object_unref(app);
    return status;
}
```

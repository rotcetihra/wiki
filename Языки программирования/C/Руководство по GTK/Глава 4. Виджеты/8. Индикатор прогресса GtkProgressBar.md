Виджет GtkProgressBar представляет индикатор прогресса некоторой работы, процесса. Для создания виджета применяется функция-конструктор `gtk_progress_bar_new()`, которая не принимает параметров.

Виджет имеет ряд свойств, которые позволяют настраивать его состояние:

- fraction: индикатор-значение прогресса
- inverted: устанавливает, будет ли индикатор выполнения двигаться в обратном порядке
- show-text: устанавливает, будет ли индикатор выполнения отображать текст в дополнение к самой полосе виджета
- text: текст (над виджетом) 

Отмечу несколько основных функций, которые могут понадобиться при работе с виджетом и, в частности, с вышеупомянутыми свойствами. Прежде всего для собственно отображения индикатора-значения прогресса на виджете применяется функция `gtk_progress_bar_set_fraction()`:

```c
void gtk_progress_bar_set_fraction (
  GtkProgressBar* pbar,
  double fraction
)
```

В качестве второго параметра передается значение типа double от 0.0 до 1.0.

Для установки текста (над виджетом) применяется функция `gtk_progress_bar_set_text()`, которая в качестве второго параметра принимает устанавливаемый текст:

```c
void gtk_progress_bar_set_text (
  GtkProgressBar* pbar,
  const char* text
)
```

Но чтобы установленный текст отображался, надо также использовать функцию `gtk_progress_bar_set_show_text()` и в качестве второго параметра передать ей TRUE:

```c
void gtk_progress_bar_set_show_text (
  GtkProgressBar* pbar,
  gboolean show_text
)
```

Если передается FALSE текст не отображается. По умолчанию текст не отображается.

Например, определим простейший индикатор прогресса:

```c
#include <gtk/gtk.h>

static void app_activate (GApplication *app, gpointer *user_data) {

    GtkWidget *window = gtk_application_window_new (GTK_APPLICATION (app));
    gtk_window_set_title (GTK_WINDOW (window), "METANIT.COM");
    gtk_window_set_default_size (GTK_WINDOW (window), 300, 250);

    // создаем прогрессбар
    GtkWidget* progressBar = gtk_progress_bar_new();
    // устанавливаем значение прогресса - 40%
    gtk_progress_bar_set_fraction(GTK_PROGRESS_BAR(progressBar), 0.4);
    // будем отображать текст
    gtk_progress_bar_set_show_text(GTK_PROGRESS_BAR(progressBar), TRUE);
    // отображаемый текст - "Progress"
    gtk_progress_bar_set_text(GTK_PROGRESS_BAR(progressBar), "Progress");

    gtk_window_set_child (GTK_WINDOW (window), progressBar);
    gtk_window_present (GTK_WINDOW (window));
}

int main (int argc, char **argv)
{
    GtkApplication *app = gtk_application_new ("com.metanit", G_APPLICATION_DEFAULT_FLAGS);
    g_signal_connect (app, "activate", G_CALLBACK (app_activate), NULL);
    int status = g_application_run (G_APPLICATION (app), argc, argv);
    g_object_unref(app);
    return status;
}
```

Определение аналогичного индикатора прогресса в XML (файл "builder.ui"):

```xml
<?xml version="1.0" encoding="UTF-8"?>
<interface>
  <object class="GtkWindow" id="window">
    <property name="title">METANIT.COM</property>
    <property name="default-width">250</property>
    <property name="default-height">200</property>
    <child>
        <object class="GtkProgressBar" id="progressBar">
          <property name="show-text">TRUE</property>
          <property name="text">Progress</property>
          <property name="fraction">0.4</property>
        </object>
    </child>
  </object>
</interface>
```

И основной файл приложения, где загружается интерфейс:

```c
#include <gtk/gtk.h>

static void app_activate (GtkApplication *app, gpointer *user_data) {

    GtkBuilder* builder = gtk_builder_new_from_file("builder.ui");
    GObject* window = gtk_builder_get_object (builder, "window");
    g_object_unref (builder);
    gtk_window_set_application (GTK_WINDOW (window), app);
    gtk_window_present (GTK_WINDOW (window));
}

int main (int argc, char **argv)
{
    GtkApplication *app = gtk_application_new ("com.metanit", G_APPLICATION_DEFAULT_FLAGS);
    g_signal_connect (app, "activate", G_CALLBACK (app_activate), NULL);
    int status = g_application_run (G_APPLICATION (app), argc, argv);
    g_object_unref(app);

    return status;
}
```

## Динамический индикатор прогресса

Но статический индикатор прогресса не очень интересен. Определим простейшее динамическое изменение прогресса индикатора:

```c
#include <gtk/gtk.h>
#include <unistd.h>

void* progress(void* arg) 
{
    GtkProgressBar* progressBar = GTK_PROGRESS_BAR(arg);
    char str[4];
    for(double i=0.0; i < 1.0; i+=0.1){
        g_print("%f\n", i);
        gtk_progress_bar_set_fraction(progressBar, i);
        // преобразуем double в строку
        snprintf(str, 4, "%f", i);
        gtk_progress_bar_set_text(GTK_PROGRESS_BAR(progressBar), str);
        sleep(1);   // задержка в 1 секунду
    }
    return NULL;
}

// при отображении окна запускаем поток
static void window_show(GtkWidget* self, gpointer *progressBar) {
    pthread_t thread1;
    pthread_create(&thread1, NULL, progress, progressBar);
}

static void app_activate (GApplication *app, gpointer *user_data) {

    GtkWidget *window = gtk_application_window_new (GTK_APPLICATION (app));
    gtk_window_set_title (GTK_WINDOW (window), "METANIT.COM");
    gtk_window_set_default_size (GTK_WINDOW (window), 300, 250);


    GtkWidget* progressBar = gtk_progress_bar_new();
    // будем отображать текст
    gtk_progress_bar_set_show_text(GTK_PROGRESS_BAR(progressBar), TRUE);

    gtk_window_set_child (GTK_WINDOW (window), progressBar);
    // устанавливаем обработчик отображения окна
    g_signal_connect (window, "show", G_CALLBACK (window_show), progressBar);
    gtk_window_present (GTK_WINDOW (window));
}

int main (int argc, char **argv)
{
    GtkApplication *app = gtk_application_new ("com.metanit", G_APPLICATION_DEFAULT_FLAGS);
    g_signal_connect (app, "activate", G_CALLBACK (app_activate), NULL);
    int status = g_application_run (G_APPLICATION (app), argc, argv);
    g_object_unref(app);
    return status;
}
```

Отмечу основные моменты. Прежде всего при запуске окна подключаем обработчик к сигналу отображения окна:

```c
g_signal_connect (window, "show", G_CALLBACK (window_show), progressBar);
```

В качестве дополнительного аргумента передаем в обработчик виджет GtkProgressBar. То есть, когда окно начнет отображаться на экране компьютера, сработает сигнал "show", и соответственно будет запущена функция обработчик `window_show()`:

```c
static void window_show(GtkWidget* self, gpointer *progressBar) {
    pthread_t thread1;
    pthread_create(&thread1, NULL, progress, progressBar);
}
```

Здесь просто запускаем новый поток, в котором выполняется функция `progress()`, в которую передается опять же виджет GtkProgressBar. В этом потоке изменяем индикатор прогресса:

```c
void* progress(void* arg) 
{
    GtkProgressBar* progressBar = GTK_PROGRESS_BAR(arg);
    char str[4];
    for(double i=0.0; i < 1.0; i+=0.1){
        g_print("%f\n", i); // логгируем на консоль
        gtk_progress_bar_set_fraction(progressBar, i);
        // преобразуем double в строку
        snprintf(str, 4, "%f", i);
        gtk_progress_bar_set_text(GTK_PROGRESS_BAR(progressBar), str);
        sleep(1);   // задержка в 1 секунду
    }
    return NULL;
}
```

В этой функции в цикле выполняем приращение переменной `i` и используем ее для установки текста и значения виджета. Чтобы процесс был более наглядным, выполняем небольшую задержку с помощью библиотечной функции `sleep()`.

В итоге при отображении окна будет запущен поток, в котором будет изменяется значение прогрессбара.

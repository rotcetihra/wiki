Виджет GtkCheckButton представляет кнопку, которая может находиться в отмеченном или неотмеченном состоянии. Для создания подобной кнопки применяется одна из функций-конструкторов:

```c
GtkWidget* gtk_check_button_new (void)
GtkWidget* gtk_check_button_new_with_label (const char* label)
GtkWidget* gtk_check_button_new_with_mnemonic (const char* label)
```

Вторая и третья функции (`gtk_check_button_new_with_label()` и `gtk_check_button_new_with_mnemonic()`) в качестве параметра принимают текст, который будет отображаться рядом с кнопкой. При этом третья функция также устанавливает мнемонику для быстрого доступа к виджету - в качестве мнемоники применяется первая буква текста.

Создадим простейший GtkCheckButton, используя вторую функцию:

```c
#include <gtk/gtk.h>

static void app_activate (GApplication *app, gpointer *user_data) {

    GtkWidget *window = gtk_application_window_new (GTK_APPLICATION (app));
    gtk_window_set_title (GTK_WINDOW (window), "METANIT.COM");
    gtk_window_set_default_size (GTK_WINDOW (window), 300, 250);

    // создаем флажок
    GtkWidget* checkBox = gtk_check_button_new_with_label("Включено");

    // размещаем виджет по верхней границе
    gtk_widget_set_valign(checkBox, GTK_ALIGN_START);

    gtk_window_set_child (GTK_WINDOW (window), checkBox);
    gtk_window_present (GTK_WINDOW (window));
}

int main (int argc, char **argv)
{
    GtkApplication *app = gtk_application_new ("com.metanit", G_APPLICATION_DEFAULT_FLAGS);
    g_signal_connect (app, "activate", G_CALLBACK (app_activate), NULL);
    int status = g_application_run (G_APPLICATION (app), argc, argv);
    g_object_unref(app);
    return status;
}
```

Определение аналогичной кнопки в XML (файл "builder.ui"):

```xml
<?xml version="1.0" encoding="UTF-8"?>
<interface>
  <object class="GtkWindow" id="window">
    <property name="title">METANIT.COM</property>
    <property name="default-width">250</property>
    <property name="default-height">200</property>
    <child>
        <object class="GtkCheckButton" id="checkBox">
          <property name="label">Включено</property>
          <property name="valign">GTK_ALIGN_START</property>
        </object>
    </child>
  </object>
</interface>
```

И основной файл приложения, где загружается интерфейс:

```c
#include <gtk/gtk.h>

static void app_activate (GtkApplication *app, gpointer *user_data) {

    GtkBuilder* builder = gtk_builder_new_from_file("builder.ui");
    GObject* window = gtk_builder_get_object (builder, "window");
    g_object_unref (builder);
    gtk_window_set_application (GTK_WINDOW (window), app);
    gtk_window_present (GTK_WINDOW (window));
}

int main (int argc, char **argv)
{
    GtkApplication *app = gtk_application_new ("com.metanit", G_APPLICATION_DEFAULT_FLAGS);
    g_signal_connect (app, "activate", G_CALLBACK (app_activate), NULL);
    int status = g_application_run (G_APPLICATION (app), argc, argv);
    g_object_unref(app);

    return status;
}
```

Стоит отметить, что мы можем динамически изменять текст кнопки с помощью функции `gtk_check_button_set_label()`:

```c
void gtk_check_button_set_label (GtkCheckButton* self, const char* label)
```

Вкратце рассмотрим функционал виджета. Основные свойства:

- child: дочерний виджет
- label: текст кнопки
- group: группа кнопки
- active: является ли кнопка отмеченной
 
Основные функции:

- `gtk_check_button_get_child`: возвращает вложенный виджет
- `gtk_check_button_set_child`: устанавливает вложенный виджет
- `gtk_check_button_get_label`: возвращает текст кнопки
- `gtk_check_button_set_label`: устанавливает текст кнопки
- `gtk_check_button_get_active`: возвращает состояние кнопки
- `gtk_check_button_set_active`: устанавливает состояние кнопки (устанавливает или сбрасывает отметку)
- `gtk_check_button_get_group`: возвращает группу кнопки
- `gtk_check_button_set_group`: устанавливает группу кнопки

Разберем вкратце основной функционал виджета.

## Выбор GtkCheckButton

Как было сказано выше, виджет GtkCheckButton может пребывать в отмеченном или неотмеченном состоянии. Для выделения кнопки применяется функция `gtk_check_button_set_active()`:

```c
void gtk_check_button_set_active (
  GtkCheckButton* self,
  gboolean setting
)
```

В качестве второго параметра передается значение TRUE (если кнопку надо отметить) или FALSE (если, наоборот, надо сбросить отметку). Для получения состояния виджета применяется функция `gtk_check_button_get_active()`:

```c
gboolean gtk_check_button_get_active (
  GtkCheckButton* self
)
```

Например, установим отметку на кнопке:

```c
#include <gtk/gtk.h>

static void app_activate (GApplication *app, gpointer *user_data) {

    GtkWidget *window = gtk_application_window_new (GTK_APPLICATION (app));
    gtk_window_set_title (GTK_WINDOW (window), "METANIT.COM");
    gtk_window_set_default_size (GTK_WINDOW (window), 300, 250);

    // создаем флажок
    GtkWidget* checkBox = gtk_check_button_new_with_label("Включено");
    // выделяем флажок
    gtk_check_button_set_active(GTK_CHECK_BUTTON(checkBox), TRUE);

    gtk_widget_set_valign(checkBox, GTK_ALIGN_START);
    gtk_window_set_child (GTK_WINDOW (window), checkBox);
    gtk_window_present (GTK_WINDOW (window));
}

int main (int argc, char **argv)
{
    GtkApplication *app = gtk_application_new ("com.metanit", G_APPLICATION_DEFAULT_FLAGS);
    g_signal_connect (app, "activate", G_CALLBACK (app_activate), NULL);
    int status = g_application_run (G_APPLICATION (app), argc, argv);
    g_object_unref(app);
    return status;
}
```

В XML для отметки кнопки надо установить свойство `active`:

```xml
<object class="GtkCheckButton" id="checkBox">
    <property name="active">1</property>
    ..........................
</object>
```

## Установка содержимого

GTK предоставляет нам виджет GtkCheckButton с некоторой базовой функциональностью. Так, метка кнопки имеет определенный шрифт и прочие характеристики. Однако, возможно, потребуется изменить эти характеристики как-то по своему. GTK позволяет изменить содержимое кнопки - в частности, его метку, которая отображает текст. Для этого предназначена функция `gtk_check_button_set_child()`:

```c
void gtk_check_button_set_child (
  GtkCheckButton* button,
  GtkWidget* child
)
```

Например, изменим метку виджета:

```c
static void app_activate (GApplication *app, gpointer *user_data) {

    GtkWidget *window = gtk_application_window_new (GTK_APPLICATION (app));
    gtk_window_set_title (GTK_WINDOW (window), "METANIT.COM");
    gtk_window_set_default_size (GTK_WINDOW (window), 300, 250);

    GtkWidget* label = gtk_label_new(NULL);
    const char *str = "active";
    const char *format = "<span font=\"15\">%s</span>";
    char *markup = g_markup_printf_escaped (format, str);
    gtk_label_set_markup (GTK_LABEL (label), markup);
    g_free(markup);

    // создаем флажок
    GtkWidget* checkBox = gtk_check_button_new();
    gtk_check_button_set_child(GTK_CHECK_BUTTON(checkBox), label);

    gtk_widget_set_valign(checkBox, GTK_ALIGN_START);
    gtk_window_set_child (GTK_WINDOW (window), checkBox);
    gtk_window_present (GTK_WINDOW (window));
}
```

Аналогичная кнопка в XML:

```xml
<object class="GtkCheckButton" id="checkBox">
    <property name="valign">GTK_ALIGN_START</property>
    <child>
        <object class="GtkLabel" id="label">
            <property name="use-markup">true</property>
            <property name="label"><span font="15">active</span></property>
        </object>
    </child>
</object>
```

## Обработка переключения кнопки

При выборе виджета GtkCheckButton или, наоборот, при снятии отметки GTK генерирует сигнал toggled:

```c
void toggled (GtkCheckButton* self, gpointer user_data)
```

Соответственно, если нам надо обработать изменение состояния виджета, то мы можем обрабатывать этот сигнал:

```c
#include <gtk/gtk.h>

const char* ON = "Включено";
const char* OFF = "Выключено";

static void checkbox_toggled (GtkCheckButton* self, gpointer user_data){
    GtkLabel* label = GTK_LABEL(user_data);
    if(gtk_check_button_get_active(self))  gtk_label_set_text (label, ON);
    else gtk_label_set_text (label, OFF);
}
static void app_activate (GApplication *app, gpointer *user_data) {

    GtkWidget *window = gtk_application_window_new (GTK_APPLICATION (app));
    gtk_window_set_title (GTK_WINDOW (window), "METANIT.COM");
    gtk_window_set_default_size (GTK_WINDOW (window), 300, 250);

    // по умолчанию кнопка отображает строку OFF
    GtkWidget* label = gtk_label_new(OFF);

    GtkWidget* checkBox = gtk_check_button_new();
    gtk_check_button_set_child(GTK_CHECK_BUTTON(checkBox), label);
    // в сигнал toggled передаем метку label
    g_signal_connect (checkBox, "toggled", G_CALLBACK (checkbox_toggled), label); 

    gtk_widget_set_valign(checkBox, GTK_ALIGN_START);
    gtk_window_set_child (GTK_WINDOW (window), checkBox);
    gtk_window_present (GTK_WINDOW (window));
}

int main (int argc, char **argv)
{
    GtkApplication *app = gtk_application_new ("com.metanit", G_APPLICATION_DEFAULT_FLAGS);
    g_signal_connect (app, "activate", G_CALLBACK (app_activate), NULL);
    int status = g_application_run (G_APPLICATION (app), argc, argv);
    g_object_unref(app);
    return status;
}
```

Для переключения текста при переключении состояния виджета определены две константы - ON и OFF. А в качестве дополнительного параметра сигналу toggled передается метка, что позволяет нам управлять ее свойствами в обработчике сигнала:

```c
g_signal_connect (checkBox, "toggled", G_CALLBACK (checkbox_toggled), label); 
```

В обработчике сигнала `toggled` - функции `checkbox_toggled` в зависимости от состояния виджета устанавливаем для ее метки определенный текст:

```c
static void checkbox_toggled (GtkCheckButton* self, gpointer user_data){
    GtkLabel* label = GTK_LABEL(user_data);
    if(gtk_check_button_get_active(self))  gtk_label_set_text (label, ON);
    else gtk_label_set_text (label, OFF);
}
```

## Создание группы радиокнопок

Виджет GtkCheckButton также позволяет создать группу радиокнопок - по сути набор кнопок, где пользователь одномоментно может выбрать только одну кнопку. Чтобы создать подобную группу кнопок, надо использовать функцию `gtk_check_button_set_group()`:

```c
void gtk_check_button_set_group (
  GtkCheckButton* self,
  GtkCheckButton* group
)
```

Функция принимает два параметра, причем оба должны представлять тип GtkCheckButton*. Первый параметр представляет кнопку, которая добавляется в группу. Второй параметр представляет кнопку, на который проецируется вся группа. Рассмотрим небольшой пример:

```c
#include <gtk/gtk.h>

static void app_activate (GApplication *app, gpointer *user_data) {

    GtkWidget *window = gtk_application_window_new (GTK_APPLICATION (app));
    gtk_window_set_title (GTK_WINDOW (window), "METANIT.COM");
    gtk_window_set_default_size (GTK_WINDOW (window), 300, 250);

    GtkWidget* box =  gtk_box_new(GTK_ORIENTATION_VERTICAL, 10);
    gtk_widget_set_halign(box, GTK_ALIGN_START); 

    // кнопка, на которую проецируется вся группа
    GtkWidget* langs = gtk_check_button_new();

    // создаем кнопки для выбора
    GtkWidget* java = gtk_check_button_new_with_label("Java");
    GtkWidget* cpp = gtk_check_button_new_with_label("C++");
    GtkWidget* python = gtk_check_button_new_with_label("Python");

    // добавляем кнопки в GtkBox
    gtk_box_append (GTK_BOX (box), java);
    gtk_box_append (GTK_BOX (box), cpp);
    gtk_box_append (GTK_BOX (box), python);

    // добавляем кнопки в группу langs
    gtk_check_button_set_group(GTK_CHECK_BUTTON(java), GTK_CHECK_BUTTON(langs));
    gtk_check_button_set_group(GTK_CHECK_BUTTON(cpp), GTK_CHECK_BUTTON(langs));
    gtk_check_button_set_group(GTK_CHECK_BUTTON(python), GTK_CHECK_BUTTON(langs));

    gtk_window_set_child (GTK_WINDOW (window), box);
    gtk_window_present (GTK_WINDOW (window));
}

int main (int argc, char **argv)
{
    GtkApplication *app = gtk_application_new ("com.metanit", G_APPLICATION_DEFAULT_FLAGS);
    g_signal_connect (app, "activate", G_CALLBACK (app_activate), NULL);
    int status = g_application_run (G_APPLICATION (app), argc, argv);
    g_object_unref(app);
    return status;
}
```

Здесь вначале создаем кнопку, с которой будет ассоциироваться группа радиокнопок:

```c
GtkWidget* langs = gtk_check_button_new();
```

Далее создаем собственно те радиокнопки, которые будут входить в эту группу и из которых мы можем выбрать одну:

```c
GtkWidget* java = gtk_check_button_new_with_label("Java");
GtkWidget* cpp = gtk_check_button_new_with_label("C++");
GtkWidget* python = gtk_check_button_new_with_label("Python");
```

Добавляем эти кнопки в контейнер GtkBox и затем добавляем их в группу langs:

```c
gtk_check_button_set_group(GTK_CHECK_BUTTON(java), GTK_CHECK_BUTTON(langs));
gtk_check_button_set_group(GTK_CHECK_BUTTON(cpp), GTK_CHECK_BUTTON(langs));
gtk_check_button_set_group(GTK_CHECK_BUTTON(python), GTK_CHECK_BUTTON(langs));
```

Таким образом, мы получим группу кнопок, где мы сможем выбрать только одну из них.

Теперь посмотрим, как мы можем отследить выбранную кнопку:

```c
#include <gtk/gtk.h>

GtkWidget* java;
GtkWidget* cpp;
GtkWidget* python;

static void selected (GtkCheckButton* self, gpointer user_data){
    // если кнопка активна (в отмеченном состоянии)
    if(gtk_check_button_get_active(self)) {
        GtkLabel* label = GTK_LABEL(user_data);
        // устанавливаем соответствующий текст на метке
        if(self==GTK_CHECK_BUTTON(cpp)) gtk_label_set_text (label, "C++");
        else if(self==GTK_CHECK_BUTTON(java)) gtk_label_set_text (label, "Java");
        else if(self==GTK_CHECK_BUTTON(python)) gtk_label_set_text (label, "Python");
    }
}
static void app_activate (GApplication *app, gpointer *user_data) {

    GtkWidget *window = gtk_application_window_new (GTK_APPLICATION (app));
    gtk_window_set_title (GTK_WINDOW (window), "METANIT.COM");
    gtk_window_set_default_size (GTK_WINDOW (window), 300, 250);

    GtkWidget* box =  gtk_box_new(GTK_ORIENTATION_VERTICAL, 10);

    gtk_widget_set_halign(box, GTK_ALIGN_START); // выравнивание по левому краю

    GtkWidget* label = gtk_label_new(NULL);

    GtkWidget* langs = gtk_check_button_new();

    java = gtk_check_button_new_with_label("Java");
    cpp = gtk_check_button_new_with_label("C++");
    python = gtk_check_button_new_with_label("Python");

    gtk_box_append (GTK_BOX (box), label);
    gtk_box_append (GTK_BOX (box), java);
    gtk_box_append (GTK_BOX (box), cpp);
    gtk_box_append (GTK_BOX (box), python);

    gtk_check_button_set_group(GTK_CHECK_BUTTON(java), GTK_CHECK_BUTTON(langs));
    gtk_check_button_set_group(GTK_CHECK_BUTTON(cpp), GTK_CHECK_BUTTON(langs));
    gtk_check_button_set_group(GTK_CHECK_BUTTON(python), GTK_CHECK_BUTTON(langs));

    // добавляем для сигнала toggled для всех кнопок обработчик
    g_signal_connect (cpp, "toggled", G_CALLBACK (selected), label);
    g_signal_connect (java, "toggled", G_CALLBACK (selected), label);
    g_signal_connect (python, "toggled", G_CALLBACK (selected), label);

    gtk_window_set_child (GTK_WINDOW (window), box);
    gtk_window_present (GTK_WINDOW (window));
}

int main (int argc, char **argv)
{
    GtkApplication *app = gtk_application_new ("com.metanit", G_APPLICATION_DEFAULT_FLAGS);
    g_signal_connect (app, "activate", G_CALLBACK (app_activate), NULL);
    int status = g_application_run (G_APPLICATION (app), argc, argv);
    g_object_unref(app);
    return status;
}
```

Здесь для большей простоты я определил все три кнопки в качестве глобальных переменных. Также в отличие от предыдущего листинга кода я добавил метку label для вывода выбранного варианта. При запуске приложения для каждой кнопки устанавливается связь сигнала toggled с обработчиком - функцией `selected()`:

```c
g_signal_connect (cpp, "toggled", G_CALLBACK (selected), label);
g_signal_connect (java, "toggled", G_CALLBACK (selected), label);
g_signal_connect (python, "toggled", G_CALLBACK (selected), label);
```

В обработчик для каждой кнопки будет передаваться метка label. В самом обработчике `selected` проверяем состояние кнопки - если она выбрана, то устанавливаем соответствующий текст на метке:

```c
static void selected (GtkCheckButton* self, gpointer user_data){
    // если кнопка активна (в отмеченном состоянии)
    if(gtk_check_button_get_active(self)) {
        GtkLabel* label = GTK_LABEL(user_data);
        // устанавливаем соответствующий текст на метке
        if(self==GTK_CHECK_BUTTON(cpp)) gtk_label_set_text (label, "C++");
        else if(self==GTK_CHECK_BUTTON(java)) gtk_label_set_text (label, "Java");
        else if(self==GTK_CHECK_BUTTON(python)) gtk_label_set_text (label, "Python");
    }
}
```

Проверка на состояние кнопки необходима, так как при выборе одной из кнопок эта функция срабатывает два раза - для кнопки, которая теряет выбор, и для кнопки которая получает выбор и становится активной.

В результате при выборе одной из кнопок метка будет отображать соответствующий текст.

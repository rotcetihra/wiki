В прошлой статье было рассмотрено создание виджета GtkColumnView, который содержал один столбец. Но, естественно, практическая польза GtkColumnView проявляется именно при работе с множеством столбцов. Поэтому рассмотрим, как определить в GtkColumnView несколько столбцов.

При добавлении столбцов в GtkColumnView следует понимать, что для каждого столбца должна использоваться своя фабрика GtkListItemFactory. Например, возьмем следующую программу:

```c
#include <gtk/gtk.h>

/******** Начало определения типа Person ************/ 
typedef struct _Person {
    GObject parent_instance;
    char *name;
    int age;
} Person;

typedef struct _PersonClass {
    GObjectClass parent_class;
} PersonClass;

#define PERSON_TYPE person_get_type()
G_DEFINE_TYPE(Person, person, G_TYPE_OBJECT)  // Определяем новый тип GObject

// Функция инициализации объекта Person
static void person_init(Person *person) {
    person->name = NULL;
    person->age = 0;
}

// Функция финализации (удаления) объекта Person
static void person_finalize(GObject *object) {
    Person *person = (Person *)object;

    if (person->name) {
        g_free(person->name);  // освобождение памяти
    }

    G_OBJECT_CLASS(person_parent_class)->finalize(object);
}
// идентификаторы свойств
enum {
    PROP_NAME=1,
    PROP_AGE,
    N_PROPERTIES
};
// определение свойств
static GParamSpec* obj_properties[N_PROPERTIES] = { NULL, };

// установка свойств
static void person_set_property(GObject* object, guint prop_id, const GValue* value, GParamSpec* pspec) {
    Person* self = (Person*)object;

    switch (prop_id) {
        case PROP_NAME:
            g_free(self->name);
            self->name = g_value_dup_string(value);
            break;
        case PROP_AGE:
            self->age = g_value_get_int(value);
            break;
        default:
            G_OBJECT_WARN_INVALID_PROPERTY_ID(object, prop_id, pspec);
            break;
    }
}

// получение свойств
static void person_get_property(GObject* object, guint prop_id, GValue* value, GParamSpec* pspec) {
    Person* self = (Person*)object;

    switch (prop_id) {
        case PROP_NAME:
            g_value_set_string(value, self->name);
            break;
        case PROP_AGE:
            g_value_set_int(value, self->age);
            break;
        default:
            G_OBJECT_WARN_INVALID_PROPERTY_ID(object, prop_id, pspec);
            break;
    }
}

// инициализация класса
static void person_class_init(PersonClass *class) {

    GObjectClass* object_class = G_OBJECT_CLASS(class);

    // переопределяем функции set_property и get_property для GObject
    object_class->set_property = person_set_property;
    object_class->get_property = person_get_property;

    // определяем свойства
    obj_properties[PROP_NAME] = g_param_spec_string(
        "name", "Name", "Person name", NULL, G_PARAM_READWRITE);
    obj_properties[PROP_AGE] = g_param_spec_int(
        "age", "Age", "Person age", 0, G_MAXINT, 0, G_PARAM_READWRITE);

    g_object_class_install_properties(object_class, N_PROPERTIES, obj_properties);

    G_OBJECT_CLASS(class)->finalize = person_finalize;
}

// Функция-конструктор для объекта Person
Person* person_new(const char *name, int age) {
    Person *person = g_object_new(PERSON_TYPE, NULL);
    person->name = g_strdup(name);  // копируем строку
    person->age = age;
    return person;
}
/******** Завершение определения типа Person ************/ 

static void setup_name (GtkListItemFactory *factory, GtkListItem *listitem)
{
    GtkWidget *label = gtk_label_new(NULL);
    gtk_widget_set_halign(label, GTK_ALIGN_START);
    gtk_list_item_set_child (listitem, label);
}
static void bind_name (GtkListItemFactory *factory, GtkListItem *listitem)
{
    GtkWidget* label = gtk_list_item_get_child(listitem);
    Person* person = (Person*) gtk_list_item_get_item (listitem);
    gtk_label_set_text (GTK_LABEL (label), person->name);
}

static void setup_age (GtkListItemFactory *factory, GtkListItem *listitem)
{
    GtkWidget *label = gtk_label_new(NULL);
    gtk_widget_set_halign(label, GTK_ALIGN_START);
    gtk_list_item_set_child (listitem, label);
}
static void bind_age (GtkListItemFactory *factory, GtkListItem *listitem)
{
    GtkWidget* label = gtk_list_item_get_child(listitem);
    Person* person = (Person*) gtk_list_item_get_item (listitem);
    char age_in_str[4]; // не больше 3 цифр + плюс нулевой символ
    // преобразуем число в строку
    snprintf(age_in_str, 4, "%d", person->age);
    gtk_label_set_text (GTK_LABEL (label), age_in_str);
}

static void app_activate (GApplication *app, gpointer *user_data) {

    GtkWidget *window = gtk_application_window_new (GTK_APPLICATION (app));
    gtk_window_set_title (GTK_WINDOW (window), "METANIT.COM");
    gtk_window_set_default_size (GTK_WINDOW (window), 300, 250);

    // список данных типа Person
    GListStore *store = g_list_store_new(PERSON_TYPE); 
    // добавляем ряд объектов
    g_list_store_append(store, person_new("Tom", 40));
    g_list_store_append(store, person_new("Bob", 46));
    g_list_store_append(store, person_new("Sam", 28));
    // определяем модель
    GtkSingleSelection* model = gtk_single_selection_new(G_LIST_MODEL(store));

    // Создание GtkColumnView
    GtkWidget* column_view = gtk_column_view_new(GTK_SELECTION_MODEL(model));
    gtk_widget_set_halign(column_view, GTK_ALIGN_FILL);

    // Создание фабрики для вывода имени
    GtkListItemFactory* name_factory = gtk_signal_list_item_factory_new();
    g_signal_connect (name_factory, "setup", G_CALLBACK (setup_name), NULL);
    g_signal_connect (name_factory, "bind", G_CALLBACK (bind_name), NULL);

    // Создание фабрики для вывода возраста
    GtkListItemFactory* age_factory = gtk_signal_list_item_factory_new();
    g_signal_connect (age_factory, "setup", G_CALLBACK (setup_age), NULL);
    g_signal_connect (age_factory, "bind", G_CALLBACK (bind_age), NULL);

    // создаем столбцы
    GtkColumnViewColumn* name_column = gtk_column_view_column_new("Name", name_factory);
    gtk_column_view_column_set_expand(name_column, TRUE);
    GtkColumnViewColumn* age_column = gtk_column_view_column_new("Age", age_factory);
    gtk_column_view_column_set_expand(age_column, TRUE);

    gtk_column_view_append_column(GTK_COLUMN_VIEW(column_view), name_column);
    gtk_column_view_append_column(GTK_COLUMN_VIEW(column_view), age_column);

    gtk_window_set_child (GTK_WINDOW (window), column_view);
    gtk_window_present (GTK_WINDOW (window));
}
int main (int argc, char **argv)
{
    GtkApplication *app = gtk_application_new ("com.metanit", G_APPLICATION_DEFAULT_FLAGS);
    g_signal_connect (app, "activate", G_CALLBACK (app_activate), NULL);
    int status = g_application_run (G_APPLICATION (app), argc, argv);
    g_object_unref(app);

    return status;
}
```

Разберем основные моменты. Прежде всего здесь данные представлены структурой Person, которая имеет поля `name` и `age`:

```c
typedef struct _Person {
    GObject parent_instance;
    char *name;
    int age;
} Person;
```

В функции `app_activate` определяем список объектов Person - объект GListStore и на его основе создаем модель:

```c
GListStore *store = g_list_store_new(PERSON_TYPE); 

g_list_store_append(store, person_new("Tom", 40));
g_list_store_append(store, person_new("Bob", 46));
g_list_store_append(store, person_new("Sam", 28));
GtkSingleSelection* model = gtk_single_selection_new(G_LIST_MODEL(store));
```

Теперь нам надо вывести эти данные в GtkColumnView. Для этого для каждого свойства структуры создаем отдельную фабрику:

```c
// Создание фабрики для вывода имени
GtkListItemFactory* name_factory = gtk_signal_list_item_factory_new();
g_signal_connect (name_factory, "setup", G_CALLBACK (setup_name), NULL);
g_signal_connect (name_factory, "bind", G_CALLBACK (bind_name), NULL);

// Создание фабрики для вывода возраста
GtkListItemFactory* age_factory = gtk_signal_list_item_factory_new();
g_signal_connect (age_factory, "setup", G_CALLBACK (setup_age), NULL);
g_signal_connect (age_factory, "bind", G_CALLBACK (bind_age), NULL);
```

Для каждой фабрики определяется своя пара обработчиков сигналов `setup` и `bind`. Например, возьмем фабрику для создания виджета для вывода имени. В обработчике сигнала `setup` для этой фабрики устанавливает виджет для вывода имени - виджет GtkLabel:

```c
static void setup_name (GtkListItemFactory *factory, GtkListItem *listitem)
{
    GtkWidget *label = gtk_label_new(NULL);
    gtk_widget_set_halign(label, GTK_ALIGN_START);
    gtk_list_item_set_child (listitem, label);
}
```

В обработчике сигнала `bind` получаем элемент модели - объект Person и выводим значение его поля `name` в созданную метку:

```c
static void bind_name (GtkListItemFactory *factory, GtkListItem *listitem)
{
    GtkWidget* label = gtk_list_item_get_child(listitem);
    Person* person = (Person*) gtk_list_item_get_item (listitem);
    gtk_label_set_text (GTK_LABEL (label), person->name);
}
```

Обработчики сигналов фабрики `age_factory` аналогичны, только в этом случае мы получаем из Person целочисленное поле `age`, которое еще надо преобразовать в строку. Затем используем обе фабрики для создания обоих столбцов и добавляем их в GtkColumnView:

```c
GtkColumnViewColumn* name_column = gtk_column_view_column_new("Name", name_factory);
GtkColumnViewColumn* age_column = gtk_column_view_column_new("Age", age_factory);

gtk_column_view_append_column(GTK_COLUMN_VIEW(column_view), name_column);
gtk_column_view_append_column(GTK_COLUMN_VIEW(column_view), age_column);
```

## Определение в xml

```xml
<object class="GtkColumnView" id="column_view">
    <property name="model">
```

При определении GtkColumnView в xml следует учитывать, что каждый столбец - элемент GtkColumnViewColumn помещается в элемент `<child>`. Внутри определения столбца можно определить фабрику. В данном случае используем фабрику GtkBuilderListItemFactory, которая использует xml для определения шаблона элемента списка. Например, определение фабрики для первого столбца:

```xml
<property name="factory">
 <object class="GtkBuilderListItemFactory">
   <property name="bytes"><![CDATA[
   <?xml version="1.0" encoding="UTF-8"?>
   <interface>
     <template class="GtkListItem">
       <property name="child">
         <object class="GtkLabel">
             <property name="xalign">0</property>
             <binding name="label">
               <lookup name="name" type="Person">
                 <lookup name="item">GtkListItem</lookup>
               </lookup>
             </binding>
           </object>
       </property>
     </template>
   </interface>
       ]]>
   </property>
 </object>
```

В шаблоне элемента определяем виджет GtkLabel, который привязывается к определенному свойству объекта Person:

```xml
<lookup name="name" type="Person">
  <lookup name="item">GtkListItem</lookup>
</lookup>
```

## Выделение строк

Для работы с выделением строк GTK предоставляет ряд функций:

- `gtk_list_box_get_selected_row`: возвращает выбранную строку или NULL, если не выбрано ни одной строки.
- `gtk_list_box_get_selected_rows`: возвращает список всех выбранных строк
- `gtk_list_box_select_all`: выбирает все строки
- `gtk_list_box_select_row`: выбирает определенную строку
- `gtk_list_box_unselect_all`: отменяет выбор всех выбранных строк
- `gtk_list_box_unselect_row`: отменяет выбор одной строки

Для выбора всех строк в функцию `gtk_list_box_get_selected_rows` передается объект GtkListBox. Для выбора одной строки в функцию `gtk_list_box_select_row()` также передается строка в виде объекта GtkListBoxRow:

```c
void gtk_list_box_select_row (GtkListBox* box, GtkListBoxRow* row)
```

Например, выбор второй строки:

```c
GtkWidget* listBox = gtk_list_box_new();

//.......................

GtkListBoxRow* row = gtk_list_box_get_row_at_index(GTK_LIST_BOX(listBox), 1);
gtk_list_box_select_row(GTK_LIST_BOX(listBox), row);
```

Другой пример - получим выделенную строку. Для этого накидаем простейшее приложение:

```c
#include <gtk/gtk.h>

GtkWidget* listBox;
GtkWidget* label;

static void select_lang(GtkButton *btn) {

    // получаем выбранную строку
    GtkListBoxRow* selectedRow = gtk_list_box_get_selected_row(GTK_LIST_BOX(listBox));
    // получаем метку из выбранной строки
    GtkWidget* rowLabel = gtk_list_box_row_get_child(selectedRow);
    // получаем текст на метке из выбранной строки
    const char* lang = gtk_label_get_text(GTK_LABEL(rowLabel));
    // устанавливаем текст на верхней метке
    gtk_label_set_text(GTK_LABEL(label), lang);
}

static void app_activate (GApplication *app, gpointer *user_data) {

    GtkWidget *window = gtk_application_window_new (GTK_APPLICATION (app));
    gtk_window_set_title (GTK_WINDOW (window), "METANIT.COM");
    gtk_window_set_default_size (GTK_WINDOW (window), 300, 250);


    listBox = gtk_list_box_new();
    const char* langs[] = {"C", "C++", "JavaScript", "Python"};
    for(size_t i=0; i < sizeof(langs)/sizeof(langs[i]); ++i){
        GtkWidget* langLabel = gtk_label_new(langs[i]);
        gtk_widget_set_halign(langLabel, GTK_ALIGN_START);
        gtk_list_box_append(GTK_LIST_BOX(listBox), langLabel);
    }

    label = gtk_label_new(NULL);
    gtk_widget_set_halign(label, GTK_ALIGN_START);
    gtk_widget_set_valign(listBox, GTK_ALIGN_START);
    
    GtkWidget* button = gtk_button_new_with_label("Select");
    gtk_widget_set_halign(button, GTK_ALIGN_START); 
    gtk_widget_set_valign(listBox, GTK_ALIGN_START);

    g_signal_connect (button, "clicked", G_CALLBACK (select_lang), NULL);

    GtkWidget* box =  gtk_box_new(GTK_ORIENTATION_VERTICAL, 10);

    gtk_box_append (GTK_BOX (box), label);
    gtk_box_append (GTK_BOX (box), button);
    gtk_box_append (GTK_BOX (box), listBox);

    gtk_window_set_child (GTK_WINDOW (window), box);
    gtk_window_present (GTK_WINDOW (window));
}

int main (int argc, char **argv)
{

    GtkApplication *app = gtk_application_new ("com.metanit", G_APPLICATION_DEFAULT_FLAGS);
    g_signal_connect (app, "activate", G_CALLBACK (app_activate), NULL);
    int status = g_application_run (G_APPLICATION (app), argc, argv);
    g_object_unref(app);

    return status;
}
```

В данном случае при нажатии на кнопку в обработчике сигнала нажатия - функции `select_lang` получаем выбранную строку и передаем ее текст на метку выше списка.

По умолчанию пользователь может выбрать только одну строку, однако с помощью функции `gtk_list_box_set_selection_mode()` можно настроить этот аспект:

```c
void gtk_list_box_set_selection_mode (GtkListBox* box, GtkSelectionMode mode)
```

Второй параметр представляет значение перечисления `GtkSelectionMode`, которое может принимать следующие значения:

- `GTK_SELECTION_NONE`: выбор невозможен. Значение: 0
- `GTK_SELECTION_SINGLE`: можно выбрать только один элемент или ноль элементов. Значение: 1
- `GTK_SELECTION_BROWSE`: можно выбрать ровно один элемент. Значение: 2
- `GTK_SELECTION_MULTIPLE`: доступен выбор множества строк. Значение: 3

Например, установим множественный выбор и выведем на консоль все выбранные строки:

```c
#include <stdio.h>
#include <gtk/gtk.h>

GtkWidget* listBox;

static void print_lang(void* data, void* user_data){
    GtkListBoxRow* selectedRow = GTK_LIST_BOX_ROW(data);
    // получаем метку из выбранной строки
    GtkWidget* rowLabel = gtk_list_box_row_get_child(selectedRow);
    // получаем текст на метке из выбранной строки
    const char* lang = gtk_label_get_text(GTK_LABEL(rowLabel));
    puts(lang);
}

static void select_lang(GtkButton *btn) {

    // получаем выбранные строки
    GList* rows = gtk_list_box_get_selected_rows(GTK_LIST_BOX(listBox));
    // применяем к каждому элементу функцию print_lang
    g_list_foreach(rows, print_lang, NULL);
    // очищаем список
    g_list_free(rows);
}

static void app_activate (GApplication *app, gpointer *user_data) {

    GtkWidget *window = gtk_application_window_new (GTK_APPLICATION (app));
    gtk_window_set_title (GTK_WINDOW (window), "METANIT.COM");
    gtk_window_set_default_size (GTK_WINDOW (window), 300, 250);

    listBox = gtk_list_box_new();
    gtk_list_box_set_selection_mode(GTK_LIST_BOX(listBox), GTK_SELECTION_MULTIPLE);
    const char* langs[] = {"C", "C++", "JavaScript", "Python"};
    for(size_t i=0; i < sizeof(langs)/sizeof(langs[i]); ++i){
        GtkWidget* langLabel = gtk_label_new(langs[i]);
        gtk_widget_set_halign(langLabel, GTK_ALIGN_START);
        gtk_list_box_append(GTK_LIST_BOX(listBox), langLabel);
    }
    
    GtkWidget* button = gtk_button_new_with_label("Select");
    gtk_widget_set_halign(button, GTK_ALIGN_START); 
    gtk_widget_set_valign(listBox, GTK_ALIGN_START);

    g_signal_connect (button, "clicked", G_CALLBACK (select_lang), NULL);

    GtkWidget* box =  gtk_box_new(GTK_ORIENTATION_VERTICAL, 10);

    gtk_box_append (GTK_BOX (box), button);
    gtk_box_append (GTK_BOX (box), listBox);


    gtk_window_set_child (GTK_WINDOW (window), box);
    gtk_window_present (GTK_WINDOW (window));
}

int main (int argc, char **argv)
{

    GtkApplication *app = gtk_application_new ("com.metanit", G_APPLICATION_DEFAULT_FLAGS);
    g_signal_connect (app, "activate", G_CALLBACK (app_activate), NULL);
    int status = g_application_run (G_APPLICATION (app), argc, argv);
    g_object_unref(app);

    return status;
}
```

Функция `gtk_list_box_get_selected_rows()` возвращает набор строк в виде контейнера GList. С помощью функции `g_list_foreach()` можно установить функцию, которая применяется к каждому элементу.

```c
void
g_list_foreach (
  GList* list,
  GFunc func,
  gpointer user_data
)
```

В нашем случае это функция `print_lang`, которая в качестве первого параметра принимает нажатую строку. Также обратите внимание, что в обработчике сигнала в конце необходимо очистить объект GList с помощью функции `g_list_free()`.

## Обработка выбора и активации строки

Чтобы строку можно было выбрать или нажать на нее, соответствующий элемент GtkListBoxRow можно пометить как активируемый или выбираемый. Для этой цели GTK предоставляет соответственно функции `gtk_list_box_row_set_activatable` и `gtk_list_box_row_set_selectable`, которые в принципе однотипны и в качестве второго параметра принимают булевое значение TRUE или FALSE:

```c
void gtk_list_box_row_set_selectable (GtkListBoxRow* row, gboolean selectable)

void gtk_list_box_row_set_activatable (GtkListBoxRow* row, gboolean activatable)
```

Если строка активируемая, и пользователь нажимает на нее, то для нее будет сгенерирован сигнал `row-activated`:

```c
void row_activated (GtkListBox* self, GtkListBoxRow* row, gpointer user_data)
```

Второй параметр представляет нажатую строку, а третий параметр - передаваемые в обработчик сигнала данные. Например, выведем текст нажатой строки в метку:

```c
#include <gtk/gtk.h>

void select_lang(GtkListBox* self, GtkListBoxRow* row, gpointer user_data){

    GtkLabel* header = GTK_LABEL(user_data);
    GtkWidget* rowLabel = gtk_list_box_row_get_child(row);
    const char* rowText = gtk_label_get_text(GTK_LABEL(rowLabel));
    gtk_label_set_text(header, rowText);
}

static void app_activate (GApplication *app, gpointer *user_data) {

    GtkWidget* window = gtk_application_window_new (GTK_APPLICATION (app));
    gtk_window_set_title (GTK_WINDOW (window), "METANIT.COM");
    gtk_window_set_default_size (GTK_WINDOW (window), 300, 250);

    GtkWidget* listBox = gtk_list_box_new();

    const char* langs[] = {"C", "Python", "C++", "JavaScript"};
    for(size_t i=0; i < sizeof(langs)/sizeof(langs[i]); ++i){
        GtkWidget* langLabel = gtk_label_new(langs[i]);
        gtk_widget_set_halign(langLabel, GTK_ALIGN_START);
        gtk_list_box_append(GTK_LIST_BOX(listBox), langLabel);
    }

    GtkWidget* label = gtk_label_new(NULL);
    gtk_widget_set_halign(label, GTK_ALIGN_START);
    gtk_widget_set_valign(listBox, GTK_ALIGN_START);
    
    g_signal_connect (listBox, "row_activated", G_CALLBACK (select_lang), label);

    GtkWidget* box =  gtk_box_new(GTK_ORIENTATION_VERTICAL, 10);

    gtk_box_append (GTK_BOX (box), label);
    gtk_box_append (GTK_BOX (box), listBox);

    gtk_window_set_child (GTK_WINDOW (window), box);
    gtk_window_present (GTK_WINDOW (window));
}

int main (int argc, char **argv)
{
    GtkApplication *app = gtk_application_new ("com.metanit", G_APPLICATION_DEFAULT_FLAGS);
    g_signal_connect (app, "activate", G_CALLBACK (app_activate), NULL);
    int status = g_application_run (G_APPLICATION (app), argc, argv);
    g_object_unref(app);

    return status;
}
```

В данном случае за обработку сигнала отвечает функция `select_lang`, в которую в качестве третьего параметра передается метка для вывода текста из активированной строки. Таким образом, текст нажатой строки отобразится в метке.

Если строка выбираемая (как это происходит по умолчанию), то строка будет помечена как выбранная, когда пользователь попытается ее выбрать, а GTK сгенерирует сигнал `row_selected`:

```c
void row_selected (GtkListBox* self, GtkListBoxRow* row, gpointer user_data)
```

Здесь аналогичный сигнал, только в этом случае пользователь может отменить выбор, и параметр `row` в этом случае будет равен `NULL`.

Тип GListStore представляет встроенную реализацию интерфейса GListModel, которая хранит все элементы в памяти. Она обеспечивает вставки, удаления и поиск по списку. Для создания хранилища GListStore применяется функция `g_list_store_new`:

```c
GListStore* g_list_store_new (GType item_type)
```

В качестве параметра функция принимает значение GType, которое представляет тип объектов хранилища. Причем тип объектов должен представлять тип GObject, то есть использовать в качестве данных произвольную структуру не получится.

GListStore имеет следующие свойства:

- `item-type`: тип элементов. Элементы должны быть подклассами GObject.
- `n-items`: количество элементов в списке

Для управления элементами для типа GListStore определен ряд функций:

- `g_list_store_append`: добавляет элемент в хранилище. Элемент должен иметь тип, указанный в свойстве `item-type`.
- `g_list_store_find`: ищет указанный элемент в хранилище, перебирая элементы до первого вхождения элемента. Если элемент не найден, то позиция не будет установлена, и этот метод вернет FALSE.
- `g_list_store_insert`: вставляет элемент в хранилище на определенную позицию.
- `g_list_store_remove`: удаляет элемент из хранилища, который находится в определенной позиции.
- `g_list_store_remove_all`: удаляет все элементы из хранилища.
- `g_list_store_sort`: сортирует элементы в хранилище.

Рассмотрим применение. Пусть у нас есть следующая программа:

```c
#include <gtk/gtk.h>

// определение структуры, которая представляет данные
typedef struct _Person {
    GObject parent_instance;
    char *name;
    int age;
} Person;

typedef struct _PersonClass {
    GObjectClass parent_class;
} PersonClass;

#define PERSON_TYPE person_get_type()
G_DEFINE_TYPE(Person, person, G_TYPE_OBJECT)  // Определяем новый тип GObject

// Функция инициализации объекта Person
static void person_init(Person *person) {
    person->name = NULL;
    person->age = 0;
}

// Функция финализации (удаления) объекта Person
static void person_finalize(GObject *object) {
    Person *person = (Person *)object;

    if (person->name) {
        g_free(person->name);  // освобождение памяти
    }

    G_OBJECT_CLASS(person_parent_class)->finalize(object);
}
// идентификаторы свойств
enum {
    PROP_NAME=1,
    PROP_AGE,
    N_PROPERTIES
};
// определение свойств
static GParamSpec* obj_properties[N_PROPERTIES] = { NULL, };


// установка свойств
static void person_set_property(GObject* object, guint prop_id, const GValue* value, GParamSpec* pspec) {
    Person* self = (Person*)object;

    switch (prop_id) {
        case PROP_NAME:
            g_free(self->name);
            self->name = g_value_dup_string(value);
            break;
        case PROP_AGE:
            self->age = g_value_get_int(value);
            break;
        default:
            G_OBJECT_WARN_INVALID_PROPERTY_ID(object, prop_id, pspec);
            break;
    }
}

// получение свойств
static void person_get_property(GObject* object, guint prop_id, GValue* value, GParamSpec* pspec) {
    Person* self = (Person*)object;

    switch (prop_id) {
        case PROP_NAME:
            g_value_set_string(value, self->name);
            break;
        case PROP_AGE:
            g_value_set_int(value, self->age);
            break;
        default:
            G_OBJECT_WARN_INVALID_PROPERTY_ID(object, prop_id, pspec);
            break;
    }
}

// инициализация класса
static void person_class_init(PersonClass *class) {

    GObjectClass* object_class = G_OBJECT_CLASS(class);

    // переопределяем функции set_property и get_property для GObject
    object_class->set_property = person_set_property;
    object_class->get_property = person_get_property;

    // определяем свойства
    obj_properties[PROP_NAME] = g_param_spec_string(
        "name", "Name", "Person name", NULL, G_PARAM_READWRITE);
    obj_properties[PROP_AGE] = g_param_spec_int(
        "age", "Age", "Person age", 0, G_MAXINT, 0, G_PARAM_READWRITE);

    g_object_class_install_properties(object_class, N_PROPERTIES, obj_properties);

    G_OBJECT_CLASS(class)->finalize = person_finalize;
}

// Функция-конструктор для объекта Person
Person* person_new(const char *name, int age) {
    Person *person = g_object_new(PERSON_TYPE, NULL);
    person->name = g_strdup(name);  // копируем строку
    person->age = age;
    return person;
}


static void app_activate (GtkApplication *app, gpointer *user_data) {

    GtkBuilder *builder = gtk_builder_new_from_file("builder.ui");
    GObject* window = gtk_builder_get_object (builder, "window");
    // получаем объект GtkListView
    GObject* listview = gtk_builder_get_object (builder, "listview");
    g_object_unref (builder);

     // Регистрируем кастомный тип Person
    GListStore *store = g_list_store_new(PERSON_TYPE); 
    // добавляем ряд объектов
    Person *tom = person_new("Tom", 40);
    g_list_store_append(store, tom);

    Person *bob = person_new("Bob", 46);
    g_list_store_append(store, bob);

    Person *sam = person_new("Sam", 28);
    g_list_store_append(store, sam);
    // устанавливаем модель
    GtkSingleSelection* model = gtk_single_selection_new(G_LIST_MODEL(store));
    gtk_list_view_set_model(GTK_LIST_VIEW(listview), GTK_SELECTION_MODEL(model));

    gtk_window_set_application (GTK_WINDOW (window), app);
    gtk_window_present (GTK_WINDOW (window));
}

int main (int argc, char **argv)
{
    GtkApplication *app = gtk_application_new ("com.metanit", G_APPLICATION_DEFAULT_FLAGS);
    g_signal_connect (app, "activate", G_CALLBACK (app_activate), NULL);
    int status = g_application_run (G_APPLICATION (app), argc, argv);
    g_object_unref(app);

    return status;
}
```

Здесь определена структура Person, которая реализует функционал GObject. Благодаря этому ее можно использовать в качестве типа объектов в GListStore.

В функции `app_activate` создаем объект GListStore, который будет хранить объекты типа Person:

```c
GListStore *store = g_list_store_new(PERSON_TYPE); 
```

Затем для примера добавляем несколько объектов Person:

```c
Person *tom = person_new("Tom", 40);
g_list_store_append(store, tom);

Person *bob = person_new("Bob", 46);
g_list_store_append(store, bob);

Person *sam = person_new("Sam", 28);
g_list_store_append(store, sam);
```

И далее на основе GListStore определяем модель для нашего списка GtkListView:

```c
GtkSingleSelection* model = gtk_single_selection_new(G_LIST_MODEL(store));
gtk_list_view_set_model(GTK_LIST_VIEW(listview), GTK_SELECTION_MODEL(model));
```

Теперь определим файл `builder.ui`, который программа использует в качестве определения интерфейса:

```xml
<?xml version="1.0" encoding="UTF-8"?>
<interface>
  <object class="GtkWindow" id="window">
    <property name="visible">True</property>
    <property name="title">METANIT.COM</property>
    <property name="default-width">250</property>
    <property name="default-height">200</property>
    <child>
        <object class="GtkListView" id="listview">
          <property name="factory">
            <object class="GtkBuilderListItemFactory">
<property name="bytes"><![CDATA[
<?xml version="1.0" encoding="UTF-8"?>
<interface>
  <template class="GtkListItem">
    <property name="child">
      <object class="GtkBox">
        <property name="orientation">GTK_ORIENTATION_HORIZONTAL</property>
        <property name="spacing">20</property>
        <child>
          <object class="GtkLabel">
          <property name="xalign">0</property>
          <binding name="label">
            <lookup name="name" type="Person">
              <lookup name="item">GtkListItem</lookup>
            </lookup>
          </binding>
        </object>
      </child>
      <child>
        <object class="GtkLabel">
          <property name="xalign">0</property>
          <binding name="label">
            <lookup name="age" type="Person">
              <lookup name="item">GtkListItem</lookup>
            </lookup>
          </binding>
        </object>
        </child>
      </object>
    </property>
  </template>
</interface>
        ]]></property>
            </object>
          </property>
        </object>
      </child>
  </object>
</interface>
```

В данном случае GtkListView в качестве фабрики применяет GtkBuilderListItemFactory, которая тут же определяет шаблон для элементов списка. В частности, каждый элемент списка будет представлять строку на основе GtkBox с двумя метками. Каждая метка берет определенное свойство объекта Person для вывода.

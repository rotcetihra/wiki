## Сортировка

Для GtkListBox имеется функционал для сортировки элементов - для этого применяется функция `gtk_list_box_set_sort_func()`:

```c
void
gtk_list_box_set_sort_func (
  GtkListBox* box,
  GtkListBoxSortFunc sort_func,
  gpointer user_data,
  GDestroyNotify destroy
)
```

Параметры функции:

- `sort_func`: функция сортировки в виде указателя типа `GtkListBoxSortFunc`:

```c
int (* GtkListBoxSortFunc) (GtkListBoxRow* row1, GtkListBoxRow* row2, gpointer user_data)
```

Функция получает две сравниваемые строки и данные, которые передаются через параметр `user_data`.

- `user_data`: данные, переданные в `sort_func`
- `destroy`: функция типа `GDestroyNotify`, которая удаляет `user_data`

Если далее потребуется отменить результаты сортировки, то применяется функция `gtk_list_box_invalidate_sort()`:

```c
void gtk_list_box_invalidate_sort (GtkListBox* box)
```

Причем сортировка идет динамически, даже если строки меняют свое содержимое. Например:

```c
#include <gtk/gtk.h>

int sort_list(GtkListBoxRow* row1, GtkListBoxRow* row2, gpointer user_data){

    GtkWidget* row1Label = gtk_list_box_row_get_child(row1);
    GtkWidget* row2Label = gtk_list_box_row_get_child(row2);
    const char* row1Text = gtk_label_get_text(GTK_LABEL(row1Label));
    const char* row2Text = gtk_label_get_text(GTK_LABEL(row2Label));
    return strcmp(row1Text, row2Text);
}

static void app_activate (GApplication *app, gpointer *user_data) {

    GtkWidget *window = gtk_application_window_new (GTK_APPLICATION (app));
    gtk_window_set_title (GTK_WINDOW (window), "METANIT.COM");
    gtk_window_set_default_size (GTK_WINDOW (window), 300, 250);

    GtkWidget* listBox = gtk_list_box_new();

    const char* langs[] = {"C", "Python", "C++", "JavaScript"};
    for(size_t i=0; i < sizeof(langs)/sizeof(langs[i]); ++i){
        GtkWidget* langLabel = gtk_label_new(langs[i]);
        gtk_widget_set_halign(langLabel, GTK_ALIGN_START);
        gtk_list_box_append(GTK_LIST_BOX(listBox), langLabel);
    }
    // устанавливаем функцию сортировки
    gtk_list_box_set_sort_func(GTK_LIST_BOX(listBox), sort_list, NULL, NULL);

    gtk_window_set_child (GTK_WINDOW (window), listBox);
    gtk_window_present (GTK_WINDOW (window));
}

int main (int argc, char **argv)
{
    GtkApplication *app = gtk_application_new ("com.metanit", G_APPLICATION_DEFAULT_FLAGS);
    g_signal_connect (app, "activate", G_CALLBACK (app_activate), NULL);
    int status = g_application_run (G_APPLICATION (app), argc, argv);
    g_object_unref(app);

    return status;
}
```

В данном случае сортировка производится в функции `sort_list`, где сравниваем текст строк с помощью встроенной библиотечной функции `strcmp`. Таким образом, строки будут автоматически сортироваться.

## Фильтрация

GtkListBox также позволяет отфильтровать элементы - для этого применяется функция `gtk_list_box_set_filter_func()`:

```c
void
gtk_list_box_set_filter_func (
  GtkListBox* box,
  GtkListBoxFilterFunc filter_func,
  gpointer user_data,
  GDestroyNotify destroy
)
```

Параметры функции:

- `filter_func`: функция фильтрации в виде указателя типа `GtkListBoxFilterFunc`:

```c
gboolean (* GtkListBoxFilterFunc) (GtkListBoxRow* row, gpointer user_data)
```

Функция получает строку и произвольные данные, которые передаются через параметр `user_data`, и проверяет строку на соответствие некоторому условию. Если строка соответствует условию, то функция возвращает `TRUE`, иначе возвращается `FALSE`.

- `user_data`: данные, переданные в `filter_func`
- `destroy`: функция типа `GDestroyNotify`, которая удаляет `user_data`

Если далее потребуется отменить результаты сортировки, то применяется функция `gtk_list_box_invalidate_filter()`:

```c
void gtk_list_box_invalidate_filter (GtkListBox* box)
```

Рассмотрим небольшой пример:

```c
#include <gtk/gtk.h>

gboolean filter_list(GtkListBoxRow* row, gpointer user_data){

    GtkWidget* rowLabel = gtk_list_box_row_get_child(row);
    const char* rowText = gtk_label_get_text(GTK_LABEL(rowLabel));
    char* term = (char*) user_data;
    return strstr(rowText, term)?TRUE: FALSE;
}

static void changed (GtkEditable* self, gpointer user_data) {

    char* text = gtk_editable_get_text(self);
    GtkListBox* listBox = GTK_LIST_BOX(user_data);
    // устанавливаем функцию фильтрации
    if(text) gtk_list_box_set_filter_func(listBox, filter_list, text, NULL);
}

static void app_activate (GApplication *app, gpointer *user_data) {

    GtkWidget *window = gtk_application_window_new (GTK_APPLICATION (app));
    gtk_window_set_title (GTK_WINDOW (window), "METANIT.COM");
    gtk_window_set_default_size (GTK_WINDOW (window), 300, 250);

    GtkWidget* box =  gtk_box_new(GTK_ORIENTATION_VERTICAL, 10);
    GtkWidget* entry = gtk_entry_new();
    GtkWidget* listBox = gtk_list_box_new();

    const char* langs[] = {"C", "Python", "C++", "C#", "JavaScript", "Java"};
    for(size_t i=0; i < sizeof(langs)/sizeof(langs[i]); ++i){
        GtkWidget* langLabel = gtk_label_new(langs[i]);
        gtk_widget_set_halign(langLabel, GTK_ALIGN_START);
        gtk_list_box_append(GTK_LIST_BOX(listBox), langLabel);
    }
    // подключаем обработчик ввода текста
    g_signal_connect (entry, "changed", G_CALLBACK (changed), listBox);

    gtk_box_append(GTK_BOX (box), entry);
    gtk_box_append(GTK_BOX (box), listBox);

    gtk_window_set_child (GTK_WINDOW (window), box);
    gtk_window_present (GTK_WINDOW (window));
}

int main (int argc, char **argv)
{
    GtkApplication *app = gtk_application_new ("com.metanit", G_APPLICATION_DEFAULT_FLAGS);
    g_signal_connect (app, "activate", G_CALLBACK (app_activate), NULL);
    int status = g_application_run (G_APPLICATION (app), argc, argv);
    g_object_unref(app);

    return status;
}
```

В данном случае интерфейс представлен элементом GtkBox, который содержит текстовое поле GtkEntry для ввода ключа фильтрации и собственно фильтруемый список GtkListBox. При каждом вводе в текстовое поле будет срабатывать обработчик сигнала `changed`.

```c
static void changed (GtkEditable* self, gpointer user_data) {

    char* text = gtk_editable_get_text(self);
    GtkListBox* listBox = GTK_LIST_BOX(user_data);
    // устанавливаем функцию фильтрации
    if(text) gtk_list_box_set_filter_func(listBox, filter_list, text, NULL);
}
```

Здесь мы получаем ссылку на переданный в обработчик объект GtkListBox, а также введенный в текстовое поле текст. И затем устанавливаем функцию фильтрации, в которую передаем введенный текст. В функции фильтрации - `filter_list` проверяем, имеет ли метка в списке введенный текст:

```c
gboolean filter_list(GtkListBoxRow* row, gpointer user_data){

    GtkWidget* rowLabel = gtk_list_box_row_get_child(row);
    const char* rowText = gtk_label_get_text(GTK_LABEL(rowLabel));
    char* term = (char*) user_data;
    return strstr(rowText, term)?TRUE: FALSE;
}
```

Через параметр `user_data` получаем введенный текст и с помощью встроенной библиотечной функции `strstr` проверяем наличие подстроки в тексте элемента списка. `strstr` возвращает указатель на найденную подстроку или NULL, поэтому в зависимости от этого результата возвращаем `TRUE` (подстрока найдена) или `FALSE` (подстрока не найдена).
